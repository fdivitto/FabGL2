SimpleTerminalOut

Self-running showcase of the VT/ANSI terminal output on a graphical display (640x480, 16 colors). The application writes text and escape sequences to fabgl::Terminal, exactly as a remote host would do.

The demo cycles through these screens, forever:

  1. Welcome: screen, viewport and terminal size, free memory
  2. ANSI colors: the 16 foreground and background colors (ESC[30m...ESC[37m, ESC[90m...ESC[97m, ...)
  3. Styles: bold, italic, underlined, blinking, inverse and their combinations, double width and double height lines (ESC#6, ESC#3, ESC#4)
  4. Mixed text and graphics: random points, lines, rectangles and ellipses drawn with fabgl::Canvas on the same screen, then the terminal scrolls over them
  5. Fast rendering: a VT/ANSI text animation
  6. Colored attributes: Terminal::setColorForAttribute() maps bold/italic/underline to colors, with and without the style

There are no controls.

What it demonstrates

  • fabgl::Terminal on top of fabgl::VGA16Controller
  • The ANSI/VT escape sequences supported by the terminal
  • Mixing terminal text and Canvas graphics on the same display

See SimpleTextTerminalOut for the same demo using the text-only VGA mode.

Hardware

  • VGA output (default pins)

See Building the examples.

33// random integer in the range [0, max)
34static inline int randomInt(int max) { return max > 0 ? (int) (esp_random() % (uint32_t) max) : 0; }
35// random integer in the range [min, max)
36static inline int randomInt(int min, int max) { return max > min ? min + (int) (esp_random() % (uint32_t) (max - min)) : min; }

60void slowPrintf(const char * format, ...)
61{
62 va_list ap;
63 va_start(ap, format);
64 int size = vsnprintf(nullptr, 0, format, ap) + 1;
65 if (size > 0) {
66 va_end(ap);
67 va_start(ap, format);
68 char buf[size + 1];
69 vsnprintf(buf, size, format, ap);
70 for (int i = 0; i < size; ++i) {
71 Terminal.write(buf[i]);
72 vTaskDelay(pdMS_TO_TICKS(25));
73 }
74 }
75 va_end(ap);
76}

153void demo4()
154{
155 Canvas cv(&DisplayController);
156 Terminal.write("\e[40;32m"); // background: black, foreground: green
157 slowPrintf("\nMixed text and graphics:\r\n");
158 slowPrintf("Points...\r\n");
159 for (int i = 0; i < 500; ++i) {
160 cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
161 cv.setPixel(randomInt(cv.getWidth()), randomInt(cv.getHeight()));
162 vTaskDelay(pdMS_TO_TICKS(15));
163 }
164 vTaskDelay(pdMS_TO_TICKS(500));
165 slowPrintf("\e[40;32mLines...\r\n");
166 for (int i = 0; i < 50; ++i) {
167 cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
168 cv.drawLine(randomInt(cv.getWidth()), randomInt(cv.getHeight()), randomInt(cv.getWidth()), randomInt(cv.getHeight()));
169 vTaskDelay(pdMS_TO_TICKS(50));
170 }
171 vTaskDelay(pdMS_TO_TICKS(500));
172 slowPrintf("\e[40;32mRectangles...\r\n");
173 for (int i = 0; i < 50; ++i) {
174 cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
175 cv.drawRectangle(randomInt(cv.getWidth()), randomInt(cv.getHeight()), randomInt(cv.getWidth()), randomInt(cv.getHeight()));
176 vTaskDelay(pdMS_TO_TICKS(50));
177 }
178 vTaskDelay(pdMS_TO_TICKS(500));
179 slowPrintf("\e[40;32mEllipses...\r\n");
180 for (int i = 0; i < 50; ++i) {
181 cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
182 cv.drawEllipse(randomInt(cv.getWidth()), randomInt(cv.getHeight()), randomInt(cv.getWidth()), randomInt(cv.getHeight()));
183 vTaskDelay(pdMS_TO_TICKS(50));
184 }
185 for (int i = 0; i < 30; ++i) {
186 Terminal.write("\e[40;32mScrolling...\r\n");
187 vTaskDelay(pdMS_TO_TICKS(250));
188 }
189 cv.clear();
190}

192void demo5()
193{
194 Terminal.write("\e[40;93m"); // background: black, foreground: yellow
195 Terminal.write("\e[2J"); // clear screen
196 slowPrintf("\e[10;56HFast Rendering");
197 slowPrintf("\e[12;50HThis is a VT/ANSI animation");
198 Terminal.write("\e[20h"); // automatic new line on
199 Terminal.write("\e[92m"); // light-green
200 Terminal.enableCursor(false);
201 for (int j = 0; j < 4; ++j) {
202 for (int i = 0; i < sizeof(vt_animation); ++i) {
203 Terminal.write(vt_animation[i]);
204 if (vt_animation[i] == 0x1B && vt_animation[i + 1] == 0x5B && vt_animation[i + 2] == 0x48)
205 vTaskDelay(pdMS_TO_TICKS(120)); // pause 100ms every frame
206 }
207 }
208 Terminal.enableCursor(true);
209 Terminal.write("\e[20l"); // automatic new line off
210}
