SoundChipSimulator

Plays a catalog of retro sound effects generated by a small software "sound chip" (ADSR envelope + frequency sweep) built on top of fabgl::SoundGenerator. By Carles Oriol.

The name of each effect is printed on screen while it plays: shot, boom, pong, pic, tuc, piu, fiu fiu whistle, fiu fiu serpentina, motor, then musical scales (escala, escala_inc with the three frequency sweep modes, escalax) and chords (sequential, overlapping and strummed). The sequence repeats forever. There are no controls.

The sound descriptor

Each effect is a playsounddata structure:

{ attack_ms, decay_ms, sustain (0-127), release_ms, wave, volume (0-127), duration_ms,
freq_start, freq_end, modfreq }
  • wave: WAVE_SQUARE, WAVE_SINE, WAVE_TRIANGLE, WAVE_SAW, WAVE_NOISE
  • modfreq: how the frequency moves from freq_start to freq_end: MODFREQ_NONE, MODFREQ_TO_END (whole duration), MODFREQ_TO_RELEASE (until the release phase), MODFREQ_TO_SUSTAIN (during attack and decay)

playSound() plays a sound asynchronously (in its own task, so several sounds can overlap); syncPlaySound() waits until it ends. Copy these two functions to add the same effects to your games: they are used by DirectVGA_ElMuro and ClassicRacer.

What it demonstrates

  • fabgl::SoundGenerator with waveform generators attached and detached at run time
  • Envelopes and frequency sweeps done by changing volume and frequency of a generator while it plays

Hardware

  • VGA output (default pins)
  • Audio output on GPIO25

75void iPlaySound( void *pvParameters )
76{
77 playsounddata psd = *(playsounddata *)pvParameters;
78
79 WaveformGenerator *pwave = nullptr;
80 if ( psd.wave == WAVE_SQUARE) pwave = new SquareWaveformGenerator();
81 if ( psd.wave == WAVE_SINE) pwave = new SineWaveformGenerator();
82 if ( psd.wave == WAVE_TRIANGLE) pwave = new TriangleWaveformGenerator();
83 if ( psd.wave == WAVE_SAW) pwave = new SawtoothWaveformGenerator();
84 if ( psd.wave == WAVE_NOISE) pwave = new NoiseWaveformGenerator();
85
86 int sustainVolume = psd.sustain * psd.volume / 127;
87
88 soundGenerator.attach( pwave);
89 pwave->setVolume( ((psd.attack == 0) ? ( (psd.decay != 0) ? psd.volume : sustainVolume) : 0) );
90 pwave->setFrequency( psd.freq_start );
91 pwave->enable(true );
92
93 long startTime = (uint32_t)(esp_timer_get_time() / 1000);
94
95 while ( (uint32_t)(esp_timer_get_time() / 1000) - startTime < psd.durationms)
96 {
97 long current = (uint32_t)(esp_timer_get_time() / 1000) - startTime;
98
99 if ( current < psd.attack ) // ATTACK VOLUME
100 pwave->setVolume( (psd.volume * current) / psd.attack );
101 else if ( current > psd.attack && current < (psd.attack + psd.decay)) // DECAY VOLUME
102 pwave->setVolume( mapValue( current - psd.attack, 0, psd.decay, psd.volume, sustainVolume ) );
103 else if ( current > psd.durationms - psd.release) // RELEASE VOLUME
104 pwave->setVolume( mapValue( current - (psd.durationms - psd.release), 0, psd.release, sustainVolume, 0 ) );
105 else
106 pwave->setVolume( sustainVolume );
107
108
109 if ( psd.modfreq != MODFREQ_NONE)
110 {
111 int maxtime = psd.durationms;
112 if (psd.modfreq == MODFREQ_TO_RELEASE)
113 maxtime = (psd.durationms - psd.release); // until release
114 else if (psd.modfreq == MODFREQ_TO_SUSTAIN)
115 maxtime = (psd.attack + psd.decay); // until sustain
116
117 int f = ((current > maxtime) ? psd.freq_end : (mapValue(current, 0, maxtime, psd.freq_start, psd.freq_end)));
118 pwave->setFrequency ( f ) ;
119 }
120 vTaskDelay(1);
121 }
122
123 soundGenerator.detach( pwave);
124 pwave->enable( false );
125 delete pwave;
126
127 vTaskDelete( NULL );
128}

131void playSound( playsounddata ps )
132{
133 // Now set up two tasks to run independently.
134 xTaskCreatePinnedToCore( iPlaySound, "iPlaySound",
135 4096, // This stack size can be checked & adjusted by reading the Stack Highwater
136 &ps, // sound as parameters
137 PLAY_SOUND_PRIORITY, // Priority, with 3 (configMAX_PRIORITIES - 1) being the highest, and 0 being the lowest.
138 NULL,
139 SOUND_TASK_CORE);
140}
