MouseStudio

Diagnostic tool for a PS/2 mouse. It has no video output: everything is printed on the serial console.

Every movement packet received from the mouse is printed as X/Y/wheel deltas and button states, and the mouse parameters (sample rate, resolution, scaling) can be changed on the fly to see their effect.

What it demonstrates

  • fabgl::PS2Controller default configuration (keyboard on port 0, mouse on port 1)
  • fabgl::Mouse low-level API: deltaAvailable(), getNextDelta(), identify(), reset(), setSampleRate(), setResolution(), setScaling()
  • Detection of wheel and 5-button mice

Hardware

  • PS/2 mouse on port 1 (CLK = GPIO26, DATA = GPIO27)
  • USB serial connection (115200 baud)

Usage

Open a serial monitor on the board port (for example idf.py -p PORT monitor). Move the mouse and press its buttons:

deltaX = 3 deltaY = -1 deltaZ = 0 leftBtn = 0 midBtn = 0 rightBtn = 0

Commands (single characters typed in the serial monitor):

Key Command
h Print the help, chip info and device identification
r Reset the mouse
s Next sample rate: 10, 20, 40, 60, 80, 100, 200 samples/s (cyclic, default 100)
e Next resolution: 1, 2, 4, 8 counts/mm (cyclic, default 4)
c Toggle scaling 1:1 / 1:2

See Building the examples.

67void printInfo()
68{
69 auto mouse = PS2Controller.mouse();
70
71 if (mouse->isMouseAvailable()) {
72 printf("%s", "Device Id = ");
73 switch (mouse->identify()) {
74 case PS2DeviceType::OldATKeyboard:
75 printf("%s", "\"Old AT Keyboard\"");
76 break;
77 case PS2DeviceType::MouseStandard:
78 printf("%s", "\"Standard Mouse\"");
79 break;
80 case PS2DeviceType::MouseWithScrollWheel:
81 printf("%s", "\"Mouse with scroll wheel\"");
82 break;
83 case PS2DeviceType::Mouse5Buttons:
84 printf("%s", "\"5 Buttons Mouse\"");
85 break;
86 case PS2DeviceType::MF2KeyboardWithTranslation:
87 printf("%s", "\"MF2 Keyboard with translation\"");
88 break;
89 case PS2DeviceType::M2Keyboard:
90 printf("%s", "\"MF2 keyboard\"");
91 break;
92 default:
93 printf("%s", "\"Unknown\"");
94 break;
95 }
96 } else
97 printf("%s", "Mouse Error!");
98 printf("%s", "\n");
99}

126void loop()
127{
128 auto mouse = PS2Controller.mouse();
129
130 if (Console.available() > 0) {
131 char c = Console.read();
132 switch (c) {
133 case 'h':
134 printHelp();
135 break;
136 case 'r':
137 mouse->reset();
138 printInfo();
139 break;
140 case 's':
141 ++currentSampleRateIndex;
142 if (currentSampleRateIndex == sizeof(SampleRates))
143 currentSampleRateIndex = 0;
144 if (mouse->setSampleRate(SampleRates[currentSampleRateIndex]))
145 printf("Sample rate = %d\n", SampleRates[currentSampleRateIndex]);
146 break;
147 case 'e':
148 ++currentResolution;
149 if (currentResolution == 4)
150 currentResolution = 0;
151 if (mouse->setResolution(currentResolution))
152 printf("Resolution = %d\n", 1 << currentResolution);
153 break;
154 case 'c':
155 ++currentScaling;
156 if (currentScaling == 3)
157 currentScaling = 1;
158 if (mouse->setScaling(currentScaling))
159 printf("Scaling = 1:%d\n", currentScaling);
160 break;
161 }
162 }
163
164 if (mouse->deltaAvailable()) {
165 MouseDelta mouseDelta;
166 mouse->getNextDelta(&mouseDelta);
167
168 printf("deltaX = %d deltaY = %d deltaZ = %d leftBtn = %d midBtn = %d rightBtn = %d\n",
169 mouseDelta.deltaX, mouseDelta.deltaY, mouseDelta.deltaZ,
170 mouseDelta.buttons.left, mouseDelta.buttons.middle, mouseDelta.buttons.right);
171 }
172
173}
