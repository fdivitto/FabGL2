TerminalCharSet

Prints the whole character set of the terminal font (codes 32 to 255) with its hexadecimal and decimal code, one character at a time, then starts again after 10 seconds.

Useful to see which glyph corresponds to each code (box drawing characters, accented letters, symbols) when writing programs that use the terminal.

The demo runs on its own: there are no controls.

What it demonstrates

  • fabgl::Terminal on a fabgl::VGAController (640x350)
  • ANSI escape sequences to set colors, clear the screen and position the cursor
  • Changing the colors with Terminal::setForegroundColor()

Hardware

  • VGA output (default pins)

Build

See Building the examples.

49void slowPrintf(const char * format, ...)
50{
51 va_list ap;
52 va_start(ap, format);
53 int size = vsnprintf(nullptr, 0, format, ap) + 1;
54 if (size > 0) {
55 va_end(ap);
56 va_start(ap, format);
57 char buf[size + 1];
58 vsnprintf(buf, size, format, ap);
59 for (int i = 0; i < size; ++i) {
60 Terminal.write(buf[i]);
61 vTaskDelay(pdMS_TO_TICKS(25));
62 }
63 }
64 va_end(ap);
65}

68void loop()
69{
70 Terminal.write("\e[40;32m"); // background: black, foreground: green
71 Terminal.write("\e[2J"); // clear screen
72 Terminal.write("\e[1;1H"); // move cursor to 1,1
73
74 slowPrintf("\e[37m* * FabGL - Character set show\r\n");
75 slowPrintf("\e[34m* * 2019-2026 by Fabrizio Di Vittorio - www.fabgl.com\e[32m\r\n\n");
76
77 for (int i = 32; i <= 255; ++i) {
78 Terminal.setForegroundColor(Color::Green);
79 slowPrintf("$%2x (%3d) = ", i, i);
80 Terminal.setForegroundColor(Color::Yellow);
81 slowPrintf("%c\t", (char)i);
82 if ((i - 31) % 5 == 0)
83 Terminal.write("\r\n");
84 }
85
86 vTaskDelay(pdMS_TO_TICKS(10000));
87}
