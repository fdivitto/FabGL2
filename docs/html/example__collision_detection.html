CollisionDetection

Animated space scene that shows the sprite collision detection of FabGL.

Ten spaceships and four asteroids fly across a starfield at 320x200 (64 colors) while Jupiter slowly crosses the background. When a spaceship hits another spaceship or an asteroid, the spaceship plays an explosion animation and the asteroid disappears; after the explosion both objects come back from a random screen side. Asteroids do not collide with each other.

The demo runs on its own: there are no controls.

What it demonstrates

  • fabgl::Scene: a game loop running at a fixed rate (init(), update(), collisionDetected())
  • Pixel-precise collision detection between sprites (Scene::addSprite(), updateSpriteAndDetectCollisions())
  • Sprites with several frames (spaceship + 8-frame explosion), Sprite::nextFrame()
  • Sprites drawn over a static background painted with fabgl::Canvas
  • fabgl::VGAController (64 colors) at 320x200

Hardware

  • VGA output (default pins)

See Building the examples.

58struct MyScene : public Scene {
59
60 MySprite objects_[OBJECTS_COUNT];

62 MyScene()
63 : Scene(OBJECTS_COUNT, 20, DisplayController.getViewPortWidth(), DisplayController.getViewPortHeight())
64 {
65 }

67 void startSprite(MySprite * sprite)
68 {
69 // set initial position to one of the four sides of the screen
70 int q = randomInt(4); // select beginning screen side
71 int x = (q == 0 ? randomInt(getWidth()) : (q == 1 ? getWidth() - sprite->getWidth() : (q == 2 ? randomInt(getWidth()) : 0)));
72 int y = (q == 0 ? 0 : (q == 1 ? randomInt(getHeight()) : (q == 2 ? getHeight() - sprite->getHeight() : randomInt(getHeight()))));
73 sprite->moveTo(x, y);
74 sprite->visible = true;
75 sprite->velX = - sprite->getWidth() / 4 + randomInt(sprite->getWidth() / 2);
76 sprite->velY = - sprite->getHeight() / 4 + randomInt(sprite->getHeight() / 2);
77 sprite->isAlive = true;
78 }
79
80 void init()
81 {
82 paintSpace();
83
84 // setup spaceship and asteroid sprites
85 for (int i = 0; i < OBJECTS_COUNT; ++i) {
86 MySprite * sprite = &objects_[i];
87 if (i == 0) {
88 // setup Jupiter sprite
89 sprite->addBitmap(&jupiter);
90 sprite->moveTo(0, getHeight() / 2);
91 } else if (i <= SPACESHIP_COUNT) {
92 // each spaceship sprite of type Ship has 9 bitmaps: one "spaceship" bitmap and eight "explosion[]" bitmaps
93 sprite->addBitmap(&spaceship)->addBitmap(explosion, 8);
94 sprite->isAsteroid = false;
95 addSprite(sprite); // add sprite to collision detector
96 startSprite(sprite);
97 } else {
98 // each spaceship sprite of type Asteroid has just one bitmap: "asteroid"
99 sprite->addBitmap(&asteroid);
100 sprite->isAsteroid = true;
101 addSprite(sprite); // add sprite to collision detector
102 startSprite(sprite);
103 }
104 }
105
106 DisplayController.setSprites(objects_, OBJECTS_COUNT);
107 }
108
109 void update(int updateCount)
110 {
111 for (int i = 1; i < OBJECTS_COUNT; ++i) {
112
113 MySprite * sprite = &objects_[i];
114
115 // just alive objects can be moved or checked for collisions
116 if (sprite->isAlive) {
117
118 // this object is alive, move to the next position
119 sprite->moveBy(sprite->velX, sprite->velY, DisplayController.getViewPortWidth(), DisplayController.getViewPortHeight());
120
121 // update the collision detector and check for a possible collision
122 updateSpriteAndDetectCollisions(sprite);
123
124 } else if ((updateCount % 8) == 0) {
125
126 // this sprite is dead, go to next frame (animate explosion) every 8 updates
127 sprite->nextFrame();
128 if (sprite->currentFrame == 0)
129 startSprite(sprite); // explosion animation ended, live again
130
131 }
132 }
133
134 // move Jupiter every 20 updates
135 if ((updateCount % 20) == 0)
136 objects_[0].moveBy(1, 0, DisplayController.getViewPortWidth(), DisplayController.getViewPortHeight());
137
138 DisplayController.refreshSprites();
139 }
140
141 void collisionDetected(Sprite * spriteA, Sprite * spriteB, Point collisionPoint)
142 {
143 MySprite * a = (MySprite*) spriteA;
144 MySprite * b = (MySprite*) spriteB;
145 // two asteroids cannot colide
146 if (a->isAsteroid && b->isAsteroid)
147 return;
148 // collision ok, mark both objects as dead
149 a->isAlive = b->isAlive = false;
150 // if collided object is an asteroid no explosion is shown, just hide it. Otherwise go to next frame.
151 if (a->isAsteroid)
152 a->visible = false;
153 else
154 a->nextFrame();
155 if (b->isAsteroid)
156 b->visible = false;
157 else
158 b->nextFrame();
159 }

161 void paintSpace()
162 {
163 Canvas cv(&DisplayController);
164 cv.setBrushColor(Color::Black);
165 cv.clear();
166 cv.setPenColor(Color::White);
167 cv.setBrushColor(Color::White);
168 // far stars
169 cv.setPenColor(64, 64, 64);
170 for (int i = 0; i < 400; ++i)
171 cv.setPixel(randomInt(getWidth()), randomInt(getHeight()));
172 // near stars
173 cv.setPenColor(128, 128, 128);
174 for (int i = 0; i < 50; ++i)
175 cv.setPixel(randomInt(getWidth()), randomInt(getHeight()));
176 // galaxy
177 cv.drawBitmap(80, 35, &galaxy);
178 cv.drawBitmap(220, 130, &galaxy);
179 }
