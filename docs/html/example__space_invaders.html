SpaceInvaders

A complete arcade shoot-'em-up game (320x200, 64 colors) with sprites, collision detection and sampled sounds. Watch it on YouTube.

How to play

Destroy the waves of aliens before they reach the ground. Hide behind the four shields (they are damaged by both your shots and the aliens' bombs). A mystery ship crosses the top of the screen from time to time.

Target Points
Mystery ship ? (random bonus)
Top row aliens 30
Middle row aliens 20
Bottom row aliens 10

You start with 3 lives. When a wave is destroyed the next level starts; the hi-score is kept until the board is reset.

Controls

On the intro screen press Space to play with the keyboard, or click to play with the mouse (only the available device is proposed).

Device Move Fire
Keyboard Left / Right arrows Space
Mouse Move the mouse horizontally Left button

After GAME OVER press Space (or click) to return to the intro.

What it demonstrates

  • fabgl::Scene game loop, sprites with several frames, pixel-precise collision detection
  • Drawing on the background bitmap (the shields are eroded pixel by pixel)
  • fabgl::SoundGenerator::playSamples() for sampled sound effects
  • Reading the keyboard state (isVKDown()) and the mouse (getNextDelta(), updateAbsolutePosition()) as game controllers

Hardware

  • VGA output (default pins)
  • Audio output on GPIO25
  • PS/2 keyboard on port 0 and/or PS/2 mouse on port 1

412 void update(int updateCount)
413 {
414 auto keyboard = PS2Controller.keyboard();
415 auto mouse = PS2Controller.mouse();
416
417 if (updateScore_) {
418 updateScore_ = false;
419 drawScore();
420 }
421
422 if (gameState_ == GAMESTATE_PLAYING || gameState_ == GAMESTATE_PLAYERKILLED) {
423
424 // move enemies and shoot
425 if ((updateCount % std::max(3, 21 - level_ * 2)) == 0) {
426 // handle enemy explosion
427 if (lastHitEnemy_) {
428 lastHitEnemy_->visible = false;
429 lastHitEnemy_ = nullptr;
430 }
431 // handle enemies movement
432 enemiesX_ += (-1 * (enemiesDir_ & 1) + (enemiesDir_ >> 1 & 1)) * ENEMIES_STEP_X;
433 enemiesY_ += (enemiesDir_ >> 2 & 1) * ENEMIES_STEP_Y;
434 bool touchSide = false;
435 for (int i = 0; i < ROWENEMIESCOUNT; ++i) {
436 moveEnemy(&enemiesR1_[i], enemiesX_ + i * ENEMIES_X_SPACE, enemiesY_ + 0 * ENEMIES_Y_SPACE, &touchSide);
437 moveEnemy(&enemiesR2_[i], enemiesX_ + i * ENEMIES_X_SPACE, enemiesY_ + 1 * ENEMIES_Y_SPACE, &touchSide);
438 moveEnemy(&enemiesR3_[i], enemiesX_ + i * ENEMIES_X_SPACE, enemiesY_ + 2 * ENEMIES_Y_SPACE, &touchSide);
439 moveEnemy(&enemiesR4_[i], enemiesX_ + i * ENEMIES_X_SPACE, enemiesY_ + 3 * ENEMIES_Y_SPACE, &touchSide);
440 moveEnemy(&enemiesR5_[i], enemiesX_ + i * ENEMIES_X_SPACE, enemiesY_ + 4 * ENEMIES_Y_SPACE, &touchSide);
441 }
442 switch (enemiesDir_) {
443 case ENEMY_MOV_DOWN_BEFORE_LEFT:
444 enemiesDir_ = ENEMY_MOV_RIGHT;
445 break;
446 case ENEMY_MOV_DOWN_BEFORE_RIGHT:
447 enemiesDir_ = ENEMY_MOV_LEFT;
448 break;
449 default:
450 if (touchSide)
451 enemiesDir_ = (enemiesDir_ == ENEMY_MOV_LEFT ? ENEMY_MOV_DOWN_BEFORE_LEFT : ENEMY_MOV_DOWN_BEFORE_RIGHT);
452 break;
453 }
454 // sound
455 ++enemiesSoundCount_;
456 soundGenerator.playSamples(invadersSoundSamples[enemiesSoundCount_ % 4], invadersSoundSamplesSize[enemiesSoundCount_ % 4]);
457 // handle enemies fire generation
458 if (!enemiesFire_->visible) {
459 int shottingEnemy = randomInt(enemiesAlive_);
460 for (int i = 0, a = 0; i < ROWENEMIESCOUNT * 5; ++i) {
461 if (enemies_[i].visible) {
462 if (a == shottingEnemy) {
463 enemiesFire_->x = enemies_[i].x + enemies_[i].getWidth() / 2;
464 enemiesFire_->y = enemies_[i].y + enemies_[i].getHeight() / 2;
465 enemiesFire_->visible = true;
466 break;
467 }
468 ++a;
469 }
470 }
471 }
472 }
473
474 if (gameState_ == GAMESTATE_PLAYERKILLED) {
475 // animate player explosion or restart playing other lives
476 if ((updateCount % 20) == 0) {
477 if (player_->getFrameIndex() == 1)
478 player_->setFrame(2);
479 else {
480 player_->setFrame(0);
481 gameState_ = GAMESTATE_PLAYING;
482 }
483 }
484 } else if (IntroScene::controller_ == 1 && playerVelX_ != 0) {
485 // move player using Keyboard
486 player_->x += playerVelX_;
487 player_->x = iclamp(player_->x, 0, getWidth() - player_->getWidth());
488 updateSprite(player_);
489 } else if (IntroScene::controller_ == 2 && playerAbsX_ != -1) {
490 // move player using Mouse
491 player_->x = playerAbsX_;
492 playerAbsX_ = -1;
493 updateSprite(player_);
494 }
495
496 // move player fire
497 if (playerFire_->visible) {
498 playerFire_->y -= 3;
499 if (playerFire_->y < ENEMIES_START_Y)
500 playerFire_->visible = false;
501 else
502 updateSpriteAndDetectCollisions(playerFire_);
503 }
504
505 // move enemies fire
506 if (enemiesFire_->visible) {
507 enemiesFire_->y += 2;
508 enemiesFire_->setFrame( enemiesFire_->getFrameIndex() ? 0 : 1 );
509 if (enemiesFire_->y > PLAYER_Y + player_->getHeight())
510 enemiesFire_->visible = false;
511 else
512 updateSpriteAndDetectCollisions(enemiesFire_);
513 }
514
515 // move enemy mother ship
516 if (enemyMother_->visible && enemyMother_->getFrameIndex() == 0) {
517 enemyMother_->x -= 1;
518 if (enemyMother_->x < -enemyMother_->getWidth())
519 enemyMother_->visible = false;
520 else
521 updateSprite(enemyMother_);
522 }
523
524 // start enemy mother ship
525 if ((updateCount % 800) == 0) {
526 soundGenerator.playSamples(motherShipSoundSamples, sizeof(motherShipSoundSamples), 100, 7000);
527 enemyMother_->x = getWidth();
528 enemyMother_->setFrame(0);
529 enemyMother_->visible = true;
530 }
531
532 // handle fire and movement from controller
533 if (IntroScene::controller_ == 1) {
534 // KEYBOARD controller
535 if (keyboard->isVKDown(fabgl::VK_LEFT))
536 playerVelX_ = -1;
537 else if (keyboard->isVKDown(fabgl::VK_RIGHT))
538 playerVelX_ = +1;
539 else
540 playerVelX_ = 0;
541 if (keyboard->isVKDown(fabgl::VK_SPACE) && !playerFire_->visible) // player fire?
542 fire();
543 } else if (IntroScene::controller_ == 2) {
544 // MOUSE controller
545 if (mouse->deltaAvailable()) {
546 MouseDelta delta;
547 mouse->getNextDelta(&delta);
548 mouse->updateAbsolutePosition(&delta);
549 playerAbsX_ = mouse->status().X;
550 if (delta.buttons.left && !playerFire_->visible) // player fire?
551 fire();
552 }
553 }
554 }
555
556 if (gameState_ == GAMESTATE_ENDGAME)
557 gameOver();
558
559 if (gameState_ == GAMESTATE_LEVELCHANGING)
560 levelChange();
561
562 if (gameState_ == GAMESTATE_LEVELCHANGED && esp_timer_get_time() >= pauseStart_ + 2500000) {
563 stop(); // restart from next level
564 DisplayController.removeSprites();
565 }
566
567 if (gameState_ == GAMESTATE_GAMEOVER) {
568
569 // animate player burning
570 if ((updateCount % 20) == 0)
571 player_->setFrame( player_->getFrameIndex() == 1 ? 2 : 1);
572
573 // wait for SPACE or click from mouse
574 if ((IntroScene::controller_ == 1 && keyboard->isVKDown(fabgl::VK_SPACE)) ||
575 (IntroScene::controller_ == 2 && mouse->deltaAvailable() && mouse->getNextDelta(nullptr, 0) && mouse->status().buttons.left)) {
576 stop();
577 DisplayController.removeSprites();
578 }
579
580 }
581
582 DisplayController.refreshSprites();
583 }

615 void collisionDetected(Sprite * spriteA, Sprite * spriteB, Point collisionPoint)
616 {
617 SISprite * sA = (SISprite*) spriteA;
618 SISprite * sB = (SISprite*) spriteB;
619 if (!lastHitEnemy_ && sA->type == TYPE_PLAYERFIRE && sB->type == TYPE_ENEMY) {
620 // player fire hits an enemy
621 soundGenerator.playSamples(shootSoundSamples, sizeof(shootSoundSamples));
622 sA->visible = false;
623 sB->setFrame(2);
624 lastHitEnemy_ = sB;
