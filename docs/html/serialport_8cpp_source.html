81// returns number of bytes received (in the UART rx fifo buffer)
82int SerialPort::uartGetRXFIFOCount()
83{
84 return m_dev->status.rxfifo_cnt | ((int)(m_dev->mem_cnt_status.rx_cnt) << 8);

99 while (uartGetRXFIFOCount() != 0 || m_dev->mem_rx_status.wr_addr != m_dev->mem_rx_status.rd_addr)

144 }
145
146 // CTS (in)
147 m_ctsPin = int2gpio(ctsPin);
148 if (m_ctsPin != GPIO_UNUSED) {
149 configureGPIO(m_ctsPin, GPIO_MODE_INPUT);
150 }
151
152 // DSR (in)
153 m_dsrPin = int2gpio(dsrPin);
154 if (m_dsrPin != GPIO_UNUSED) {

172// uart = 0, 1, 2
173void SerialPort::setup(int uartIndex, uint32_t baud, int dataLength, char parity, float stopBits, FlowControl flowControl, bool inverted)
174{
175 if (!m_initialized) {
176 // uart not configured, configure now
177
178 m_idx = uartIndex;
179 m_dev = (volatile uart_dev_t *) UART_BASE[m_idx];
180
181 // release the ESP-IDF UART driver, if installed on this port
182 if (uart_is_driver_installed((uart_port_t) m_idx))
183 uart_driver_delete((uart_port_t) m_idx);
184
185 switch (m_idx) {
186 case 0:
187 DPORT_SET_PERI_REG_MASK(DPORT_PERIP_CLK_EN_REG, DPORT_UART_CLK_EN);
188 DPORT_CLEAR_PERI_REG_MASK(DPORT_PERIP_RST_EN_REG, DPORT_UART_RST);
189 break;
190 case 1:
191 DPORT_SET_PERI_REG_MASK(DPORT_PERIP_CLK_EN_REG, DPORT_UART1_CLK_EN);
192 DPORT_CLEAR_PERI_REG_MASK(DPORT_PERIP_RST_EN_REG, DPORT_UART1_RST);
193 break;
194 case 2:
195 DPORT_SET_PERI_REG_MASK(DPORT_PERIP_CLK_EN_REG, DPORT_UART2_CLK_EN);
196 DPORT_CLEAR_PERI_REG_MASK(DPORT_PERIP_RST_EN_REG, DPORT_UART2_RST);
197 break;
198 }
199
200 // flush
201 uartFlushTXFIFO();
202 uartFlushRXFIFO();
203
204 // setup RX interrupt
205 WRITE_PERI_REG(UART_CONF1_REG(m_idx), (1 << UART_RXFIFO_FULL_THRHD_S) | // an interrupt for each character received
206 (2 << UART_RX_TOUT_THRHD_S) | // actually not used
207 (0 << UART_RX_TOUT_EN_S)); // timeout not enabled
208 WRITE_PERI_REG(UART_INT_ENA_REG(m_idx), (1 << UART_RXFIFO_FULL_INT_ENA_S) | // interrupt on FIFO full (1 character - see rxfifo_full_thrhd)
209 (1 << UART_FRM_ERR_INT_ENA_S) | // interrupt on frame error
210 (0 << UART_RXFIFO_TOUT_INT_ENA_S) | // no interrupt on rx timeout (see rx_tout_en and rx_tout_thrhd)
211 (1 << UART_PARITY_ERR_INT_ENA_S) | // interrupt on rx parity error
212 (1 << UART_RXFIFO_OVF_INT_ENA_S) | // interrupt on rx overflow
213 (1 << UART_BRK_DET_INT_ENA_S)); // interrupt on Break
214 WRITE_PERI_REG(UART_INT_CLR_REG(m_idx), 0xffffffff);
215 esp_intr_alloc_pinnedToCore(INTR_SRC[m_idx], 0, uart_isr, this, nullptr, CoreUsage::quietCore());
216
217 // setup FIFOs size
218 WRITE_PERI_REG(UART_MEM_CONF_REG(m_idx), (3 << UART_RX_SIZE_S) | // RX: 3 * 128 = 384 bytes (this is the max for UART2)
219 (1 << UART_TX_SIZE_S)); // TX: 1 * 128 = 128 bytes
220 }
221
222 m_flowControl = flowControl;
223
224 m_inverted = inverted;
225
226 // baud rate
227 setBaud(baud);
228
229 // frame
230 setFrame(dataLength, parity, stopBits);
231
232 // TX/RX Pin logic
233 esp_rom_gpio_connect_in_signal(m_rxPin, URXD_IN_IDX[m_idx], m_inverted);
234 esp_rom_gpio_connect_out_signal(m_txPin, UTXD_OUT_IDX[m_idx], m_inverted, false);
235
236 // Flow Control
237 WRITE_PERI_REG(UART_FLOW_CONF_REG(m_idx), 0);
238
239 m_initialized = true;
240
241 // APB Change callback (TODO?)
242 //addApbChangeCallback(this, uart_on_apb_change);
243}

246void SerialPort::setBaud(int value)
247{
248 uint32_t clk_div = (getApbFrequency() << 4) / value;
249 WRITE_PERI_REG(UART_CLKDIV_REG(m_idx), ((clk_div >> 4) << UART_CLKDIV_S) |
250 ((clk_div & 0xf) << UART_CLKDIV_FRAG_S));
251}

254void SerialPort::setFrame(int dataLength, char parity, float stopBits)
255{
256 uint32_t config0 = (1 << UART_TICK_REF_ALWAYS_ON_S) | ((dataLength - 5) << UART_BIT_NUM_S);
257 if (parity == 'E')
258 config0 |= (1 << UART_PARITY_EN_S);
259 else if (parity == 'O')
260 config0 |= (1 << UART_PARITY_EN_S) | (1 << UART_PARITY_S);
261 if (stopBits == 1.0)
262 config0 |= 1 << UART_STOP_BIT_NUM_S;
263 else if (stopBits == 1.5)
264 config0 |= 2 << UART_STOP_BIT_NUM_S;
265 else if (stopBits >= 2.0) {
266 config0 |= 1 << UART_STOP_BIT_NUM_S;
267 SET_PERI_REG_BITS(UART_RS485_CONF_REG(m_idx), UART_DL1_EN_V, 1, UART_DL1_EN_S); // additional 1 stop bit
268 if (stopBits >= 3.0)
269 SET_PERI_REG_BITS(UART_RS485_CONF_REG(m_idx), UART_DL0_EN_V, 1, UART_DL1_EN_S); // additional 1 stop bit
270 }
271 WRITE_PERI_REG(UART_CONF0_REG(m_idx), config0);
272}

275// enable/disable RX sending XON/XOFF and/or setting RTS
276void SerialPort::flowControl(bool enableRX)
277{
278 bool isrEnabled = m_dev->int_ena.rxfifo_full;
279 CLEAR_PERI_REG_MASK(UART_INT_ENA_REG(m_idx), UART_RXFIFO_FULL_INT_ENA_M);
280
281 if (enableRX) {
282 // resume RX
283 if (m_sentXOFF) {
284 if (m_flowControl == FlowControl::Software || m_flowControl == FlowControl::Hardsoft)
285 send(ASCII_XON);
286 if (m_flowControl == FlowControl::Hardware || m_flowControl == FlowControl::Hardsoft)
287 setRTSStatus(true); // assert RTS
288 isrEnabled = true;
289 m_sentXOFF = false;
290 }
291 } else {
292 // suspend RX
293 if (!m_sentXOFF) {
294 if (m_flowControl == FlowControl::Software || m_flowControl == FlowControl::Hardsoft)
295 send(ASCII_XOFF);
296 if (m_flowControl == FlowControl::Hardware || m_flowControl == FlowControl::Hardsoft)
297 setRTSStatus(false); // disable RTS
298 // note: here we don't disable FIFO interrupts. These are disabled in ISR when input queue is full
299 m_sentXOFF = true;
300 }
301 }
302
303 if (isrEnabled)
304 SET_PERI_REG_MASK(UART_INT_ENA_REG(m_idx), UART_RXFIFO_FULL_INT_ENA_M); // enable RX FIFO full interrupt
305}

308// check if TX is enabled looking for XOFF received or reading CTS
309bool SerialPort::readyToSend()
310{
311 //Serial.printf("readyToSend m_recvXOFF=%d\n", m_recvXOFF);
312 if ((m_flowControl == FlowControl::Software || m_flowControl == FlowControl::Hardsoft) && m_recvXOFF)
313 return false; // TX disabled (received XOFF)
314 if ((m_flowControl == FlowControl::Hardware || m_flowControl == FlowControl::Hardsoft) && CTSStatus() == false)
315 return false; // TX disabled (CTS=high, not active)
316 return true; // TX enabled
317}

328void SerialPort::sendBreak(bool value)
329{
330 constexpr int MATRIX_DETACH_OUT_SIG = 0x100;
331
332 while (m_dev->status.txfifo_cnt == 0x7F)
333 ;
334 if (value) {
335 esp_rom_gpio_connect_out_signal(m_txPin, MATRIX_DETACH_OUT_SIG, m_inverted, false);
336 configureGPIO(m_txPin, GPIO_MODE_OUTPUT);
337 gpio_set_level(m_txPin, 0);
338 } else {
339 esp_rom_gpio_connect_out_signal(m_txPin, UTXD_OUT_IDX[m_idx], m_inverted, false);
340 }

344void IRAM_ATTR SerialPort::uart_isr(void * arg)
345{
346 auto ser = (SerialPort*) arg;
347 auto dev = ser->m_dev;
348 auto idx = ser->m_idx;
349
350 // look for overflow or RX errors
351 if (dev->int_st.rxfifo_ovf || dev->int_st.frm_err || dev->int_st.parity_err) {
352 // save (or) error states
353 ser->m_parityError |= dev->int_st.parity_err;
354 ser->m_framingError |= dev->int_st.frm_err;
355 ser->m_overflowError |= dev->int_st.rxfifo_ovf;
356 if (ser->m_lineStatusCallback)
357 ser->m_lineStatusCallback(ser->m_callbackArgs, ser->m_parityError, ser->m_framingError, ser->m_overflowError, ser->m_breakDetected, true);
358 // reset RX-FIFO, because hardware bug rxfifo_rst cannot be used, so just flush
359 ser->uartFlushRXFIFO();
360 // clear interrupt flags
361 SET_PERI_REG_MASK(UART_INT_CLR_REG(idx), UART_RXFIFO_OVF_INT_CLR_M | UART_FRM_ERR_INT_CLR_M | UART_PARITY_ERR_INT_CLR_M);
362 return;
363 }
364
365 // Break detection
366 if (dev->int_st.brk_det) {
367 // save (or) break state
368 ser->m_breakDetected |= dev->int_st.brk_det;
369 if (ser->m_lineStatusCallback)
370 ser->m_lineStatusCallback(ser->m_callbackArgs, ser->m_parityError, ser->m_framingError, ser->m_overflowError, ser->m_breakDetected, true);
371 // clear interrupt flags
372 SET_PERI_REG_MASK(UART_INT_CLR_REG(idx), UART_BRK_DET_INT_CLR_M);
373 return;
374 }
375
376 // flow control?
377 if (ser->m_flowControl != FlowControl::None) {
378 // send XOFF/XON or set RTS looking at RX FIFO occupation
379 int count = ser->uartGetRXFIFOCount();
380 if (count > FABGLIB_TERMINAL_FLOWCONTROL_RXFIFO_MAX_THRESHOLD)
381 ser->flowControl(false); // suspend RX
382 else if (count < FABGLIB_TERMINAL_FLOWCONTROL_RXFIFO_MIN_THRESHOLD)
383 ser->flowControl(true); // resume RX
384 }
385
386 // main receive loop
387 while (ser->uartGetRXFIFOCount() != 0 || dev->mem_rx_status.wr_addr != dev->mem_rx_status.rd_addr) {
388 // look for enough room in input queue
389 if (ser->m_flowControl != FlowControl::None && !ser->m_rxReadyCallback(ser->m_callbackArgs, true)) {
390 ser->flowControl(false); // suspend RX
391 // block further interrupts
392 CLEAR_PERI_REG_MASK(UART_INT_ENA_REG(idx), UART_RXFIFO_FULL_INT_ENA_M);
393 break;
394 }
395 // get char
396 uint8_t r = READ_PERI_REG(UART_FIFO_REG(idx));
397 // XON/XOFF?
398 if ((r == ASCII_XOFF || r == ASCII_XON) && (ser->m_flowControl == FlowControl::Software || ser->m_flowControl == FlowControl::Hardsoft)) {
399 ser->m_recvXOFF = (r == ASCII_XOFF);
400 } else {
401 // add to input queue
402 ser->m_rxCallback(ser->m_callbackArgs, r, true);
403 }
404 }
405
406 // clear interrupt flag
407 SET_PERI_REG_MASK(UART_INT_CLR_REG(idx), UART_RXFIFO_FULL_INT_CLR_M);
408}

431void SerialPortTerminalConnector::connect(SerialPort * serialPort, Terminal * terminal)
432{
433 m_serialPort = serialPort;
434 m_serialPort->setCallbacks(this, rxReadyCallback, rxCallback);
435
436 m_terminal = terminal;
437 m_terminal->connectKeyboard();
438
439 m_terminal->onReceive = [&](uint8_t c) {
440 // resume RX?
441 if (!m_serialPort->readyToReceive() && m_terminal->availableForWrite())
442 m_serialPort->flowControl(true);
443 };
444
445 m_terminal->onReadyToSend = [&](bool * readyToSend) {
446 // suspend/resume TX?
447 if (*readyToSend)
448 *readyToSend = m_serialPort->readyToSend();
449 };
450
451 m_terminal->onSend = [&](uint8_t c) {
452 m_serialPort->send(c);
453 };
454}
