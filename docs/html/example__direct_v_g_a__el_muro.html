DirectVGA_ElMuro

"El Muro": a small brick-breaker game rendered entirely with direct scanline drawing (fabgl::VGADirectController, 320x200, no frame buffer), with sound effects.

Contributed by Carles Oriol (author of ClassicRacer).

What it demonstrates

  • fabgl::VGADirectController: the whole game screen (bricks, ball, paddle, borders, lives) is generated line by line inside an IRAM_ATTR callback
  • Game logic synchronized to the vertical retrace through a task notification
  • fabgl::SoundGenerator with a SquareWaveformGenerator for simple sound effects
  • Reusable game controller classes (controllers.h) for keyboard, mouse and GPIO joysticks

Hardware

  • VGA output (default pins)
  • Audio output on GPIO25 (optional)
  • PS/2 keyboard on port 0 and/or PS/2 mouse on port 1

How to play

Break all the 96 bricks by bouncing the ball with the paddle. You have 3 spare paddles (shown at the top right); when you lose them all, or when all the bricks are destroyed, the game restarts. The bounce angle depends on where the ball hits the paddle.

Input Action
Mouse movement Move the paddle
Left / Right arrows Move the paddle
Left mouse button or right Shift Launch the ball (it also launches by itself after 4 seconds)

78void IRAM_ATTR drawScanline(void * arg, uint8_t * dest, int scanLine)
79{
80 auto fgcolor = DisplayController.createRawPixel(RGB222(3, 3, 3));
81 auto white = DisplayController.createRawPixel(RGB222(3, 3, 3));
82 auto black = DisplayController.createRawPixel(RGB222(0, 0, 0));
83 auto bgcolor = DisplayController.createRawPixel(RGB222(0, 0, 0));
84
85 if( scanLine >= 32 )
86 {
87 if(scanLine < 32+16)
88 {
89 bgcolor = DisplayController.createRawPixel(RGB222(1, 0, 0));
90 fgcolor = DisplayController.createRawPixel(RGB222(3, 0, 0));
91 }
92 else if(scanLine < 32+16+16)
93 {
94 bgcolor = DisplayController.createRawPixel(RGB222(1, 0, 1));
95 fgcolor = DisplayController.createRawPixel(RGB222(3, 0, 3));
96 }
97 else if(scanLine < 32+16+16+16)
98 {
99 bgcolor = DisplayController.createRawPixel(RGB222(0, 1, 0));
100 fgcolor = DisplayController.createRawPixel(RGB222(0, 3, 0));
101 }
102 else if(scanLine < 32+16+16+16+16)
103 {
104 bgcolor = DisplayController.createRawPixel(RGB222(1, 1, 0));
105 fgcolor = DisplayController.createRawPixel(RGB222(3, 3, 0));
106 }
107 }
108
109 auto width = DisplayController.getScreenWidth();
110
111 // fill line with background color
112 if( scanLine < TOPMARGIN)
113 memset(dest+8, fgcolor, width-16);
114 else
115 {
116 memset(dest, black, 16);
117 memset(dest+319-15, black, 16);
118 memset(dest+16, bgcolor, width-32);
119 for( int ncompta = 8; ncompta < SIDEMARGIN; ncompta++)
120 {
121 VGA_PIXELINROW(dest, ncompta) = white;
122 VGA_PIXELINROW(dest, 319-ncompta) = white;
123 }
124 }
125
126 // fill object with foreground color
127 if (scanLine >= objIntY - objSize / 2 && scanLine <= objIntY + objSize / 2)
128 {
129 for (int col = objIntX - objSize / 2; col < objIntX + objSize / 2; ++col)
130 VGA_PIXELINROW(dest, col) = fgcolor;
131 }
132
133 if (scanLine >= 184 && scanLine < 192 )
134 {
135 for (int col = paddlePos - 16 ; col < paddlePos + 16; ++col)
136 VGA_PIXELINROW(dest, col) = white;
137 }
138
139 // drawBricks
140 if( scanLine >= 32 && scanLine < (32+16+16+16+16) && scanLine % 8 != 0)
141 for( int x = 0; x < 12*3*8; x++)
142 {
143 if( x%24 != 0 && x%24 != 23 && brickMap[(scanLine-32)/8][x/24])
144 VGA_PIXELINROW(dest, x+16) = fgcolor;
145 }
146
147 if( scanLine >= 10 && scanLine <= 13 )
148 for (int p = 0; p < nPaddles; p++)
149 for( int x = 0; x < 10; x++)
150 VGA_PIXELINROW(dest, 300 - x - p*14) = white;
151
152 if (scanLine == DisplayController.getScreenHeight() - 1) {
153 // signal end of screen
154 vTaskNotifyGiveFromISR(mainTaskHandle, NULL);
155 }
156}

194void loop()
195{
196 if( (uint32_t)(esp_timer_get_time() / 1000) - offGame > 4000)
197 offGame = false;
198
199 // check brick collision
200 static int oldbricky = -1;
201 int brickx = (objIntX-16)/24;
202 int bricky = (objIntY-32)/8;
203
204 if( bricky >= 0 && bricky <8 && brickx < 12 && brickMap[bricky][brickx] )
205 {
206 brickMap[bricky][brickx] = false;
207 nBricks--;
208 objDir = (( bricky != oldbricky ) ? 2 : 1) * M_PI - objDir;
209 playSoundPic();
210 }
211
212 if( nBricks == 0) resetGame();
213
214 // test collision with borders and bounce changing direction
215 if (objX < objSize / 2 +SIDEMARGIN|| objX > DisplayController.getScreenWidth() - objSize / 2-SIDEMARGIN)
216 {
217 objDir = M_PI - objDir;
218 playSoundPong();
219 }
220 else if (objY < objSize / 2 +TOPMARGIN )
221 {
222 objDir = 2 * M_PI - objDir;
223 playSoundPong();
224 }
225
226 double sdir = sin(objDir);
227
228 // check paddle collision
229 if ( !offGame && objX > paddlePos -(16+objSize/2) && objX < paddlePos +(16+objSize/2) && objY > 184-objSize/2 && objY< 192 && sdir >= 0)
230 {
231 objDir = 2 * M_PI - objDir;
232 objDir += (objX - ((double)paddlePos))*((M_PI/2.0)/32.0);
233 playSoundTuc();
234 }
235
236 if( !offGame )
237 {
238 sdir = sin(objDir);
239 // calculate new coordinates
240 objX += objVel * cos(objDir);
241 objY += objVel * sdir;
242 }
243 else
244 objX = paddlePos +8;
245
246 if(objY < 7) objY = 8;
247 if( objX < 16 ) objX = 16 ;
248 if( objX > 319-16 ) objX = 319-16;
249
250 // Ball lost
251 if(objY > 200)
252 {
253 playSoundOut();
254 offGame = (uint32_t)(esp_timer_get_time() / 1000);
255 nPaddles--;
256 if( nPaddles == -1)
257 resetGame();
258 else
259 {
260 objY = INITIALY;
261 objDir = INITIALDIR;
262 }
263 }
264
265 // convert object coordinate to integer
266 objIntX = objX;
267 objIntY = objY;
268
269 oldbricky = bricky;
270
271 cMouse.update();
272 paddlePos += cMouse.getDX();
273 if( cKeysArrows.isLeft() ) paddlePos -= 8;
274 if( cKeysArrows.isRight() ) paddlePos += 8;
275 if( cMouse.isButtonA() || cKeysArrows.isButtonA() )
276 offGame = false;
277
278 if( paddlePos < 32) paddlePos = 32;
279 if( paddlePos > 319-32) paddlePos = 319-32;
280
281 // wait for vertical sync
282 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
283}
