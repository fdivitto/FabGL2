53bool MCP23S17::begin(int MISO, int MOSI, int CLK, int CS, int CSActiveState, int host)
54{
55 // defaults
56 int defMISO = 35;
57 int defMOSI = 12;
58 int defCLK = 14;
59 int defCS = -1;
60 switch (getChipPackage()) {
61 case ChipPackage::ESP32PICOD4:
62 // setup for TTGO VGA32
63 defMISO = 2;
64 defMOSI = 12;
65 break;
66 case ChipPackage::ESP32D0WDQ5:
67 // setup for FabGL compatible board
68 defCS = 13;
69 if (CSActiveState == -1)
70 CSActiveState = 1;
71 break;
72 default:
73 break;
74 }
75
76 if (MISO == -1)
77 MISO = defMISO;
78 if (MOSI == -1)
79 MOSI = defMOSI;
80 if (CS == -1)
81 CS = defCS;
82 if (CLK == -1)
83 CLK = defCLK;
84
85 m_MISO = int2gpio(MISO);
86 m_MOSI = int2gpio(MOSI);
87 m_CLK = int2gpio(CLK);
88 m_CS = int2gpio(CS);
89 m_SPIHost = (spi_host_device_t) host;
90
91 bool r = SPIBegin(CSActiveState) && initDevice(0);
92 if (!r)
93 end();
94 return r;
95}

99// - select bank 0
100// - enable hardware address
101bool MCP23S17::initDevice(uint8_t hwAddr)
102{
103 bool r = false;
104 if (hwAddr < MCP_MAXDEVICES) {
105 m_IOCON[hwAddr] = MCP_IOCON_SEQOP | MCP_IOCON_HAEN;
106 writeReg(MCP_IOCON, m_IOCON[hwAddr], hwAddr);
107 r = readReg(MCP_IOCON, hwAddr) == m_IOCON[hwAddr];
108 }
109 return r;
110}

119bool MCP23S17::SPIBegin(int CSActiveState)
120{
121 spi_bus_config_t busconf = { }; // zero init
122 busconf.mosi_io_num = m_MOSI;
123 busconf.miso_io_num = m_MISO;
124 busconf.sclk_io_num = m_CLK;
125 busconf.quadwp_io_num = -1;
126 busconf.quadhd_io_num = -1;
127 busconf.data4_io_num = -1;
128 busconf.data5_io_num = -1;
129 busconf.data6_io_num = -1;
130 busconf.data7_io_num = -1;
131 busconf.flags = SPICOMMON_BUSFLAG_MASTER;
132 auto r = spi_bus_initialize(m_SPIHost, &busconf, MCP_DMACHANNEL);
133 if (r == ESP_OK || r == ESP_ERR_INVALID_STATE) { // ESP_ERR_INVALID_STATE, maybe spi_bus_initialize already called
134 spi_device_interface_config_t devconf = { }; // zero init
135 devconf.mode = 0;
136 devconf.clock_speed_hz = MCP_SPI_FREQ;
137 devconf.spics_io_num = (m_CS == GPIO_UNUSED ? -1 : m_CS); // ESP-IDF requires -1 when CS is not used
138 devconf.flags = (CSActiveState == 1 ? SPI_DEVICE_POSITIVE_CS : 0);
139 devconf.queue_size = 1;
140 r = spi_bus_add_device(m_SPIHost, &devconf, &m_SPIDevHandle);
141 if (r != ESP_OK && !FileBrowser::mountedSDCard())
142 spi_bus_free(m_SPIHost);
143 return r == ESP_OK;
144 }
145 return false;
146}

149void MCP23S17::SPIEnd()
150{
151 if (m_SPIDevHandle) {
152 spi_bus_remove_device(m_SPIDevHandle);
153 m_SPIDevHandle = nullptr;
154 if (FileBrowser::mountedSDCard()) {
155 if (getChipPackage() == ChipPackage::ESP32D0WDQ5) {
156 fabgl::configureGPIO(m_CS, GPIO_MODE_OUTPUT);
157 gpio_set_level(m_CS, 1);
158 }
159 } else {
160 spi_bus_free(m_SPIHost);
161 }
162 }
163}

166void MCP23S17::writeReg(uint8_t addr, uint8_t value, uint8_t hwAddr)
167{
168 spi_device_acquire_bus(m_SPIDevHandle, portMAX_DELAY);
169
170 uint8_t txdata[3] = { (uint8_t)(0b01000000 | (hwAddr << 1)), addr, value };
171 spi_transaction_t ta = { };
172 ta.flags = 0;
173 ta.length = 24;
174 ta.rxlength = 0;
175 ta.rx_buffer = nullptr;
176 ta.tx_buffer = txdata;
177 spi_device_transmit(m_SPIDevHandle, &ta);
178
179 spi_device_release_bus(m_SPIDevHandle);
180}

183uint8_t MCP23S17::readReg(uint8_t addr, uint8_t hwAddr)
184{
185 spi_device_acquire_bus(m_SPIDevHandle, portMAX_DELAY);
186
187 uint8_t txdata[3] = { (uint8_t)(0b01000001 | (hwAddr << 1)), addr };
188 uint8_t rxdata[3] = { 0 };
189 spi_transaction_t ta = { };
190 ta.flags = 0;
191 ta.length = 24;
192 ta.rxlength = 24;
193 ta.rx_buffer = rxdata;
194 ta.tx_buffer = txdata;
195 uint8_t r = 0;
196 if (spi_device_transmit(m_SPIDevHandle, &ta) == ESP_OK)
197 r = rxdata[2];
198
199 spi_device_release_bus(m_SPIDevHandle);
200
201 return r;
202}

222uint16_t MCP23S17::readReg16(uint8_t addr, uint8_t hwAddr)
223{
224 spi_device_acquire_bus(m_SPIDevHandle, portMAX_DELAY);
225
226 uint8_t txdata[4] = { (uint8_t)(0b01000001 | (hwAddr << 1)), addr };
227 uint8_t rxdata[4] = { 0 };
228 spi_transaction_t ta = { };
229 ta.flags = 0;
230 ta.length = 32;
231 ta.rxlength = 32;
232 ta.rx_buffer = rxdata;
233 ta.tx_buffer = txdata;
234 uint16_t r = 0;
235 if (spi_device_transmit(m_SPIDevHandle, &ta) == ESP_OK)
236 r = rxdata[2] | (rxdata[3] << 8);
237
238 spi_device_release_bus(m_SPIDevHandle);
239
240 return r;

262void MCP23S17::configureGPIO(int gpio, MCPDir dir, bool pullup, uint8_t hwAddr)
263{
264 uint8_t mask = MCP_GPIO2MASK(gpio);
265 // direction
266 uint8_t reg = MCP_GPIO2REG(MCP_IODIR, gpio);
267 if (dir == MCPDir::Input)
268 writeReg(reg, readReg(reg, hwAddr) | mask, hwAddr);
269 else
270 writeReg(reg, readReg(reg, hwAddr) & ~mask, hwAddr);
271 // pull-up
272 reg = MCP_GPIO2REG(MCP_GPPU, gpio);
273 writeReg(reg, (readReg(reg, hwAddr) & ~mask) | ((int)pullup * mask), hwAddr);
274}

277void MCP23S17::writeGPIO(int gpio, bool value, uint8_t hwAddr)
278{
279 uint8_t olat = readReg(MCP_GPIO2REG(MCP_OLAT, gpio), hwAddr);
280 uint8_t mask = MCP_GPIO2MASK(gpio);
281 uint8_t reg = MCP_GPIO2REG(MCP_OLAT, gpio);
282 writeReg(reg, value ? olat | mask : olat & ~mask, hwAddr);
283}

292void MCP23S17::enableInterrupt(int gpio, MCPIntTrigger trigger, bool defaultValue, uint8_t hwAddr)
293{
294 uint8_t mask = MCP_GPIO2MASK(gpio);
295 // set interrupt trigger
296 if (trigger == MCPIntTrigger::DefaultChange) {
297 // interrupt triggered when value is different than "defaultValue)
298 writeReg(MCP_GPIO2REG(MCP_INTCON, gpio), readReg(MCP_GPIO2REG(MCP_INTCON, gpio), hwAddr) | mask, hwAddr);
299 writeReg(MCP_GPIO2REG(MCP_DEFVAL, gpio), (readReg(MCP_GPIO2REG(MCP_DEFVAL, gpio), hwAddr) & ~mask) | ((int)defaultValue * mask), hwAddr);
300 } else {
301 // interrupt triggered when value is different than previous value
302 writeReg(MCP_GPIO2REG(MCP_INTCON, gpio), readReg(MCP_GPIO2REG(MCP_INTCON, gpio), hwAddr) & ~mask, hwAddr);
303 }
304 // enable interrupt
305 writeReg(MCP_GPIO2REG(MCP_GPINTEN, gpio), readReg(MCP_GPIO2REG(MCP_GPINTEN, gpio), hwAddr) | mask, hwAddr);
306}

316void MCP23S17::writePort(int port, void const * buffer, size_t length, uint8_t hwAddr)
317{
318 // - disable sequential mode
319 // - select bank 1 (to avoid switching between A and B registers)
320 writeReg(MCP_IOCON, m_IOCON[hwAddr] | MCP_IOCON_SEQOP | MCP_IOCON_BANK);
321
322 spi_device_acquire_bus(m_SPIDevHandle, portMAX_DELAY);
323
324 spi_transaction_ext_t ta = { };
325 ta.command_bits = 8;
326 ta.address_bits = 8;
327 ta.base.cmd = 0b01000000 | (hwAddr << 1); // write
328 ta.base.addr = MCP_BNK1_OLAT + port * 0x10;
329 ta.base.flags = SPI_TRANS_VARIABLE_CMD | SPI_TRANS_VARIABLE_ADDR;
330 ta.base.length = 16 + 8 * length;
331 ta.base.rxlength = 0;
332 ta.base.rx_buffer = nullptr;
333 ta.base.tx_buffer = buffer;
334 spi_device_polling_transmit(m_SPIDevHandle, (spi_transaction_t*) &ta);
335
336 spi_device_release_bus(m_SPIDevHandle);
337
338 // restore IOCON
339 writeReg(MCP_BNK1_IOCON, m_IOCON[hwAddr]);
340}
