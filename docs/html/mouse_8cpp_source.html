91bool Mouse::reset()
92{
93 if (s_quickCheckHardware) {
94 m_mouseAvailable = send_cmdReset();
95 } else {
96 // tries up to three times for mouse reset
97 for (int i = 0; i < 3; ++i) {
98 m_mouseAvailable = send_cmdReset();
99 if (m_mouseAvailable)
100 break;
101 vTaskDelay(500 / portTICK_PERIOD_MS);
102 }
103 // give the time to the device to be fully initialized
104 vTaskDelay(200 / portTICK_PERIOD_MS);
105 }
106
107 // negotiate compatibility and default parameters
108 if (m_mouseAvailable) {
109 // try Intellimouse (three buttons + scroll wheel, 4 bytes packet)
110 if (send_cmdSetSampleRate(200) && send_cmdSetSampleRate(100) && send_cmdSetSampleRate(80) && identify() == PS2DeviceType::MouseWithScrollWheel) {
111 // Intellimouse ok!
112 m_mouseType = Intellimouse;
113 }
114
115 setSampleRate(60);
116
117 // make sure data reporting is enabled (identify() enables it, but it is not called when the Intellimouse
118 // negotiation fails, leaving the mouse silent)
119 send_cmdEnableScanning();
120 }
121
122 return m_mouseAvailable;
123}

150// Mouse packet format:
151// byte 0:
152// bit 0 = Left Button
153// bit 1 = Right Button
154// bit 2 = Middle Button
155// bit 3 = Always 1
156// bit 4 = X sign bit
157// bit 5 = Y sign bit
158// bit 6 = X overflow
159// bit 7 = Y overflow
160// byte 1:
161// X movement
162// byte 2:
163// Y movement
164// byte 3:
165// Z movement
166bool Mouse::decodeMousePacket(MousePacket * mousePacket, MouseDelta * delta)
167{
168 // the bit 4 of first byte must be always 1
169 if ((mousePacket->data[0] & 8) == 0)
170 return false;
171
172 m_prevStatus = m_status;
173
174 // decode packet
175 m_status.buttons.left = (mousePacket->data[0] & 0x01 ? 1 : 0);
176 m_status.buttons.middle = (mousePacket->data[0] & 0x04 ? 1 : 0);
177 m_status.buttons.right = (mousePacket->data[0] & 0x02 ? 1 : 0);
178 if (delta) {
179 delta->deltaX = (int16_t)(mousePacket->data[0] & 0x10 ? 0xFF00 | mousePacket->data[1] : mousePacket->data[1]);
180 delta->deltaY = (int16_t)(mousePacket->data[0] & 0x20 ? 0xFF00 | mousePacket->data[2] : mousePacket->data[2]);
181 delta->deltaZ = (int8_t)(getPacketSize() > 3 ? mousePacket->data[3] : 0);
182 delta->overflowX = (mousePacket->data[0] & 0x40 ? 1 : 0);
183 delta->overflowY = (mousePacket->data[0] & 0x80 ? 1 : 0);
184 delta->buttons = m_status.buttons;
185 }
186
187 return true;
188}

191// Changing parameters while the mouse is streaming data may fail (the command collides with data packets,
192// the mouse may wait forever for the parameter byte). So data reporting is suspended, the command is retried
193// on failure and finally data reporting is enabled again.
194bool Mouse::sendParameterCommand(bool (Mouse::*cmd)(int), int value)
195{
196 PS2DeviceLock deviceLock(this);
197 bool ret = false;
198 for (int i = 0; i < 3 && !ret; ++i) {
199 send_cmdDisableScanning();
200 ret = (this->*cmd)(value);
201 }
202 bool e = false;
203 for (int i = 0; i < 3 && !e; ++i)
204 e = send_cmdEnableScanning();
205 return ret;
206}

234void Mouse::setupAbsolutePositioner(int width, int height, bool createAbsolutePositionsQueue, BitmappedDisplayController * updateDisplayController, uiApp * app)
235{
236 if (m_area != Size(width, height)) {
237 m_area = Size(width, height);
238 m_status.X = width >> 1;
239 m_status.Y = height >> 1;
240 }
241 m_status.wheelDelta = 0;
242 m_status.buttons.left = 0;
243 m_status.buttons.middle = 0;
244 m_status.buttons.right = 0;
245 m_prevStatus = m_status;
246
247 m_updateDisplayController = updateDisplayController;
248
249 m_uiApp = app;
250
251 if (createAbsolutePositionsQueue && m_absoluteQueue == nullptr) {
252 m_absoluteQueue = xQueueCreate(FABGLIB_MOUSE_EVENTS_QUEUE_SIZE, sizeof(MouseStatus));
253 }
254
255 if (m_updateDisplayController) {
256 // setup initial position
257 m_updateDisplayController->setMouseCursorPos(m_status.X, m_status.Y);
258 }
259
260 m_absoluteUpdate = (m_updateDisplayController || createAbsolutePositionsQueue || m_uiApp);
261}

276void Mouse::updateAbsolutePosition(MouseDelta * delta)
277{
278 const int maxDeltaTimeUS = 500000; // after 0.5s doesn't consider acceleration
279
280 int dx = delta->deltaX;
281 int dy = delta->deltaY;
282 int dz = delta->deltaZ;
283
284 int64_t now = esp_timer_get_time();
285 int deltaTime = (int) (now - m_prevDeltaTime); // time in microseconds
286
287 if (deltaTime < maxDeltaTimeUS) {
288
289 // calculate movement acceleration
290 if (dx != 0 || dy != 0) {
291 int deltaDist = isqrt(dx * dx + dy * dy); // distance in mouse points
292 float vel = (float)deltaDist / deltaTime; // velocity in mousepoints/microsecond
293 float newVel = vel + m_movementAcceleration * vel * vel; // new velocity
294 int newDeltaDist = newVel * deltaTime; // new distance
295 dx = dx * newDeltaDist / deltaDist;
296 dy = dy * newDeltaDist / deltaDist;
297 }
298
299 // calculate wheel acceleration
300 if (dz != 0) {
301 int deltaDist = abs(dz); // distance in wheel points
302 float vel = (float)deltaDist / deltaTime; // velocity in mousepoints/microsecond
303 float newVel = vel + m_wheelAcceleration * vel * vel; // new velocity
304 int newDeltaDist = newVel * deltaTime; // new distance
305 dz = dz * newDeltaDist / deltaDist;
306 }
307
308 }
309
310 m_status.X = tclamp((int)m_status.X + dx, 0, m_area.width - 1);
311 m_status.Y = tclamp((int)m_status.Y - dy, 0, m_area.height - 1);
312 m_status.wheelDelta = dz;
313 m_prevDeltaTime = now;
314}

317void Mouse::mouseUpdateTask(void * arg)
318{
319 constexpr int MAX_TIME_BETWEEN_DATA_US = 500000; // maximum time between data composing a delta frame
320
321 Mouse * mouse = (Mouse*) arg;
322
323 int64_t prevDataTime = 0;
324
325 while (true) {
326
327 MousePacket mousePacket;
328 int mousePacketLen = 0;
329 while (mousePacketLen < mouse->getPacketSize()) {
330 int r = mouse->getData(-1);
331 if (mouse->parityError() || mouse->syncError()) {
332 mousePacketLen = 0;
333 continue;
334 }
335 int64_t now = esp_timer_get_time();
336 if (mousePacketLen > 0 && prevDataTime > 0 && (now - prevDataTime) > MAX_TIME_BETWEEN_DATA_US) {
337 // too much time elapsed since last byte, start a new delta
338 mousePacketLen = 0;
339 }
340 if (r > -1) {
341 mousePacket.data[mousePacketLen++] = r;
342 prevDataTime = now;
343 }
344 }
345
346 if (mouse->m_absoluteUpdate) {
347 MouseDelta delta;
348 if (mouse->decodeMousePacket(&mousePacket, &delta)) {
349 mouse->updateAbsolutePosition(&delta);
350
351 // VGA Controller
352 if (mouse->m_updateDisplayController)
353 mouse->m_updateDisplayController->setMouseCursorPos(mouse->m_status.X, mouse->m_status.Y);
354
355 // queue (if you need availableStatus() or getNextStatus())
356 if (mouse->m_absoluteQueue) {
357 xQueueSendToBack(mouse->m_absoluteQueue, &mouse->m_status, 0);
358 }
359
360 if (mouse->m_uiApp) {
361 // generate uiApp events
362 if (mouse->m_prevStatus.X != mouse->m_status.X || mouse->m_prevStatus.Y != mouse->m_status.Y) {
363 // X and Y movement: UIEVT_MOUSEMOVE
364 uiEvent evt = uiEvent(nullptr, UIEVT_MOUSEMOVE);
365 evt.params.mouse.status = mouse->m_status;
366 evt.params.mouse.changedButton = 0;
367 mouse->m_uiApp->postEvent(&evt);
368 }
369 if (mouse->m_status.wheelDelta != 0) {
370 // wheel movement: UIEVT_MOUSEWHEEL
371 uiEvent evt = uiEvent(nullptr, UIEVT_MOUSEWHEEL);
372 evt.params.mouse.status = mouse->m_status;
373 evt.params.mouse.changedButton = 0;
374 mouse->m_uiApp->postEvent(&evt);
375 }
376 if (mouse->m_prevStatus.buttons.left != mouse->m_status.buttons.left) {
377 // left button: UIEVT_MOUSEBUTTONDOWN, UIEVT_MOUSEBUTTONUP
378 uiEvent evt = uiEvent(nullptr, mouse->m_status.buttons.left ? UIEVT_MOUSEBUTTONDOWN : UIEVT_MOUSEBUTTONUP);
379 evt.params.mouse.status = mouse->m_status;
380 evt.params.mouse.changedButton = 1;
381 mouse->m_uiApp->postEvent(&evt);
382 }
383 if (mouse->m_prevStatus.buttons.middle != mouse->m_status.buttons.middle) {
384 // middle button: UIEVT_MOUSEBUTTONDOWN, UIEVT_MOUSEBUTTONUP
385 uiEvent evt = uiEvent(nullptr, mouse->m_status.buttons.middle ? UIEVT_MOUSEBUTTONDOWN : UIEVT_MOUSEBUTTONUP);
386 evt.params.mouse.status = mouse->m_status;
387 evt.params.mouse.changedButton = 2;
388 mouse->m_uiApp->postEvent(&evt);
389 }
390 if (mouse->m_prevStatus.buttons.right != mouse->m_status.buttons.right) {
391 // right button: UIEVT_MOUSEBUTTONDOWN, UIEVT_MOUSEBUTTONUP
392 uiEvent evt = uiEvent(nullptr, mouse->m_status.buttons.right ? UIEVT_MOUSEBUTTONDOWN : UIEVT_MOUSEBUTTONUP);
393 evt.params.mouse.status = mouse->m_status;
394 evt.params.mouse.changedButton = 3;
395 mouse->m_uiApp->postEvent(&evt);
396 }
397 }
398
399 }
400
401 } else {
402
403 xQueueOverwrite(mouse->m_receivedPacket, &mousePacket);
404
405 }
406
407 }
408}
