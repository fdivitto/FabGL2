// random integer in the range [0, max)
static inline int randomInt(int max) { return max > 0 ? (int) (esp_random() % (uint32_t) max) : 0; }
// random integer in the range [min, max)
static inline int randomInt(int min, int max) { return max > min ? min + (int) (esp_random() % (uint32_t) (max - min)) : min; }

void setup()
{

Terminal.begin(&DisplayController);
//Terminal.setLogStream(Console); // DEBUG ONLY (requires "fabgl::UARTStream Console;" and "Console.begin();")

void slowPrintf(const char * format, ...)
{
va_list ap;
va_start(ap, format);
int size = vsnprintf(nullptr, 0, format, ap) + 1;
if (size > 0) {
va_end(ap);
va_start(ap, format);
char buf[size + 1];
vsnprintf(buf, size, format, ap);
for (int i = 0; i < size; ++i) {
Terminal.write(buf[i]);
vTaskDelay(pdMS_TO_TICKS(25));
}
}
va_end(ap);
}

void demo4()
{
Canvas cv(&DisplayController);
Terminal.write("\e[40;32m"); // background: black, foreground: green
slowPrintf("\nMixed text and graphics:\r\n");
slowPrintf("Points...\r\n");
for (int i = 0; i < 500; ++i) {
cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
cv.setPixel(randomInt(cv.getWidth()), randomInt(cv.getHeight()));
vTaskDelay(pdMS_TO_TICKS(15));
}
vTaskDelay(pdMS_TO_TICKS(500));
slowPrintf("\e[40;32mLines...\r\n");
for (int i = 0; i < 50; ++i) {
cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
cv.drawLine(randomInt(cv.getWidth()), randomInt(cv.getHeight()), randomInt(cv.getWidth()), randomInt(cv.getHeight()));
vTaskDelay(pdMS_TO_TICKS(50));
}
vTaskDelay(pdMS_TO_TICKS(500));
slowPrintf("\e[40;32mRectangles...\r\n");
for (int i = 0; i < 50; ++i) {
cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
cv.drawRectangle(randomInt(cv.getWidth()), randomInt(cv.getHeight()), randomInt(cv.getWidth()), randomInt(cv.getHeight()));
vTaskDelay(pdMS_TO_TICKS(50));
}
vTaskDelay(pdMS_TO_TICKS(500));
slowPrintf("\e[40;32mEllipses...\r\n");
for (int i = 0; i < 50; ++i) {
cv.setPenColor(randomInt(256), randomInt(256), randomInt(256));
cv.drawEllipse(randomInt(cv.getWidth()), randomInt(cv.getHeight()), randomInt(cv.getWidth()), randomInt(cv.getHeight()));
vTaskDelay(pdMS_TO_TICKS(50));
}
for (int i = 0; i < 30; ++i) {
Terminal.write("\e[40;32mScrolling...\r\n");
vTaskDelay(pdMS_TO_TICKS(250));
}
cv.clear();
}
void demo5()
{
Terminal.write("\e[40;93m"); // background: black, foreground: yellow
Terminal.write("\e[2J"); // clear screen
slowPrintf("\e[10;56HFast Rendering");
slowPrintf("\e[12;50HThis is a VT/ANSI animation");
Terminal.write("\e[20h"); // automatic new line on
Terminal.write("\e[92m"); // light-green
Terminal.enableCursor(false);
for (int j = 0; j < 4; ++j) {
for (int i = 0; i < sizeof(vt_animation); ++i) {
Terminal.write(vt_animation[i]);
if (vt_animation[i] == 0x1B && vt_animation[i + 1] == 0x5B && vt_animation[i + 2] == 0x48)
vTaskDelay(pdMS_TO_TICKS(120)); // pause 100ms every frame
}
}
Terminal.enableCursor(true);
Terminal.write("\e[20l"); // automatic new line off
}
