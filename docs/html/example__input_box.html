InputBox

Guided tour of fabgl::InputBox: ready-made modal dialogs (messages, menus, text input, selections, progress bars, file and folder browsers) that need just one function call each, without writing a GUI.

The demo shows the dialogs one after the other (640x350 window on a 640x480 screen), then restarts:

  1. Welcome message, closed automatically after 10 seconds (setAutoOK()): the countdown is shown in the button
  2. Progress bar with an Abort button
  3. Simple menu built from a ;-separated string, and a second menu built from a StringList
  4. Items selection with extra buttons: Add (asks a value with a text input), Remove (asks a confirmation) and Options, a button with a drop-down of sub-items (Edit, Restore, Advanced)
  5. Quiz: a single-choice selection with a Submit button
  6. Text input: asks your name and greets you
  7. Folder browser and file selector on the SD card (/SD) or, if no SD card is present, on the internal SPIFFS partition (/Flash, formatted on first use)
  8. WiFi configuration wizard: scans the networks (progress box), lets you pick one from a menu, asks the password (masked text input) and connects, showing the result
  9. Restart message

Controls

Every dialog can be used with the mouse or the keyboard:

Key Action
Enter Default button (OK / Submit)
Esc Cancel
Up / Down Move the selection in menus and lists
Tab Move the focus to the next control

What it demonstrates

  • InputBox::message(), messageFmt(), menu(), select(), textInput(), progressBox(), folderBrowser(), fileSelector(), setupButton() (extra and drop-down buttons), setAutoOK()
  • fabgl::StringList, fabgl::WiFiScanner, fabgl::WiFiConnection, FileBrowser::mountSDCard() / mountSPIFFS()
  • Custom painting of the background (InputBox::onPaint)

Hardware

  • VGA output (default pins)
  • PS/2 keyboard on port 0 and/or PS/2 mouse on port 1
  • Optional SD card

Build

The example ships a partitions.csv with a SPIFFS partition. See Building the examples.

41void loop()
42{
43 // initializes input box
44 InputBox ib;
45 ib.begin(VGA_640x480_60Hz, 640, 350);
46 ib.setBackgroundColor(RGB888(0, 0, 0));
47
48 ib.onPaint = [&](Canvas * canvas) {
49 canvas->selectFont(&fabgl::FONT_SANSERIF_8x16);
50 canvas->setPenColor(RGB888(255, 255, 0));
51 canvas->drawText(85, 5, "InputBox Demo - www.fabgl.com - by Fabrizio Di Vittorio");
52 };
53
54 // setup automatic OK after 10 seconds
55 ib.setAutoOK(10);
56
57
58 ib.message("Welcome!", "Welcome to FabGL InputBox demo!");
59
60
62 // Example of progress bar
63 InputResult r = ib.progressBox("Example of Progress Bar", "Abort", true, 200, [&](fabgl::ProgressForm * form) {
64 for (int i = 0; i <= 100; ++i) {
65 if (!form->update(i, "Index is %d/100", i))
66 break;
67 vTaskDelay(pdMS_TO_TICKS(40));
68 }
69 vTaskDelay(pdMS_TO_TICKS(800));
70 });
71 if (r == InputResult::Cancel)
72 ib.message("", "Operation Aborted");
73
74
76 // Example of simple menu (items from separated strings)
77 int s = ib.menu("Simple Menu", "Click on one item", "Item number zero;Item number one;Item number two;Item number three");
78 ib.messageFmt("", nullptr, "OK", "You have selected item %d", s);
79
80
82 // Example of simple menu (items from StringList)
83 fabgl::StringList list;
84 list.append("Option Zero");
85 list.append("Option One");
86 list.append("Option Two");
87 list.append("Option Three");
88 list.select(1, true);
89 s = ib.menu("Menu", "Click on an item", &list);
90 ib.messageFmt("", nullptr, "OK", "You have selected item %d", s);
91
92
94 // Example of options selection box with OK button (items from separated strings)
95 list.clear();
96 list.append("Item 0");
97 list.append("Item 1");
98 list.append("Item 2");
99 for (bool loop = true; loop; ) {
100 ib.setupButton(0, "Add");
101 ib.setupButton(1, "Remove");
102 ib.setupButton(2, "Options", "Edit;Restore;Advanced", 50);
103 ib.setMinButtonsWidth(60);
104 auto r = ib.select("Items selection", "Select an item", &list, "Cancel", "OK");
105 switch (r) {
106 // OK button
107 case InputResult::Enter:
108 ib.messageFmt(nullptr, nullptr, "OK", "You have selected item %d", list.getFirstSelected());
109 loop = false;
110 break;
111 // add new item button
112 case InputResult::ButtonExt0:
113 {
114 char value[32] = "";
115 if (ib.textInput("New Item", "Please enter item value", value, 31) == InputResult::Enter) {
116 list.takeStrings();
117 list.append(value);
118 }
119 break;
120 }
121 // remove item button
122 case InputResult::ButtonExt1:
123 if (list.getFirstSelected() > -1 && ib.message("Please confirm", "Remove Item?", "No", "Yes") == InputResult::Enter)
124 list.remove(list.getFirstSelected());
125 break;
126 // button with subitems (Edit / Restore / Advanced)
127 case InputResult::ButtonExt2:
128 switch (ib.selectedSubItem()) {
129 // Edit sub-button
130 case 0:
131 ib.message(nullptr, "Edit - not implented!");
132 break;
133 // Restore
134 case 1:
135 ib.message(nullptr, "Restore - not implented!");
136 break;
137 // Advanced
138 case 2:
139 ib.message(nullptr, "Advanced - not implented!");
140 break;
141 }
142 break;
143 // cancel
144 default:
145 loop = false;
146 break;
147 }
148 }
149
151 // Example of options selection box with OK button (items from StringList)
152 fabgl::StringList quiz;
153 quiz.append("A) Connecting to the network");
154 quiz.append("B) Creating new user accounts");
155 quiz.append("C) Providing power to the computer");
156 quiz.append("D) Connecting smartphones and other peripherals");
157 if (ib.select("QUIZ", "What is an Ethernet port used for?", &quiz, nullptr, "Submit") == InputResult::Enter) {
158 ib.message("Result", quiz.selected(0) ? "That's right, great job!" : "Sorry, that's not correct");
159 }
160
161
163 // Example of text input box
164 char name[32] = "";
165 if (ib.textInput("Asking name", "What's your name?", name, 31) == InputResult::Enter)
166 ib.messageFmt("", nullptr, "OK", "Nice to meet you %s!", name);
167
168
170 // File Browser
171 char const * path = nullptr;
172 if (FileBrowser::mountSDCard(false, "/SD"))
173 path = "/SD";
174 else if (FileBrowser::mountSPIFFS(true, "/Flash"))
175 path = "/Flash";
176 if (path) {
177 ib.folderBrowser("Folder Browser", path);
178 }
179
180
182 // File Select
183 if (path) {
184 char filename[16] = "";
185 char directory[32];
186 strcpy(directory, path);
187 if (ib.fileSelector("File Select", "Filename: ", directory, sizeof(directory) - 1, filename, sizeof(filename) - 1) == InputResult::Enter) {
188 ib.messageFmt("", nullptr, "OK", "Folder = %s, File = %s", directory, filename);
189 }
190 FileBrowser::unmountSDCard();
191 FileBrowser::unmountSPIFFS();
192 }
193
194
196 // Example WiFi connection wizard
197 if (ib.message("WiFi Configuration", "Configure WiFi?", "No", "Yes") == InputResult::Enter) {
198
199 // yes, scan for networks showing a progress dialog box
200 int networksCount = 0;
201 fabgl::WiFiScanner scanner;
202 ib.progressBox("", nullptr, false, 200, [&](fabgl::ProgressForm * form) {
203 form->update(0, "Scanning WiFi networks...");
204 if (scanner.scan(16))
205 networksCount = scanner.count();
206 });
207
208 // are there available WiFi?
209 if (networksCount > 0) {
210
211 // yes, show a selectable list
212 fabgl::StringList list;
213 for (int i = 0; i < networksCount; ++i)
214 list.appendFmt("%s (%d dBm)", (char const *) scanner.get(i)->ssid, scanner.get(i)->rssi);
215 int s = ib.menu("WiFi Configuration", "Please select a WiFi network", &list);
216
217 // user selected something?
218 if (s > -1) {
219
220 // yes, ask for WiFi password
221 char psw[32] = "";
222 if (ib.textInput("WiFi Configuration", "Insert WiFi password", psw, 31, "Cancel", "OK", true) == InputResult::Enter) {
223
224 // user pressed OK, connect to WiFi...
225 char SSID[50];
226 strlcpy(SSID, (char const *) scanner.get(s)->ssid, sizeof(SSID));
227 bool connected = false;
228 static fabgl::WiFiConnection wifi;
229 ib.progressBox("", nullptr, true, 200, [&](fabgl::ProgressForm * form) {
230 wifi.connect(SSID, psw, 0); // just start connection
231 for (int i = 0; i < 32 && wifi.state() != fabgl::WiFiConnectionState::Connected; ++i) {
232 if (!form->update(i * 100 / 32, "Connecting to %s...", SSID))
233 break;
234 vTaskDelay(pdMS_TO_TICKS(500));
235 }
236 connected = (wifi.state() == fabgl::WiFiConnectionState::Connected);
237 });
238
239 // show to user the connection state
240 ib.message("", connected ? "Connection succeeded" : "Connection failed");
241
242 }
243 }
244 } else {
245 // there is no WiFi
246 ib.message("", "No WiFi network found!");
247 }
248 }
249
250 ib.message("Restart", "Ok, press OK to restart!");
251
252
253 // finalizes input box
254 ib.end();
255
256}
