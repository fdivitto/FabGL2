KeyboardScanCodes

Diagnostic tool that shows the raw scancodes sent by a PS/2 keyboard.

Every byte received from the keyboard is printed in hexadecimal, one key event per line (prefix bytes E0 and F0 are kept on the same line as the code they belong to). The output goes both to the VGA screen (80x25 text mode) and to the serial console.

What it demonstrates

  • fabgl::VGATextController (text-only VGA mode, very small memory footprint) with a fabgl::Terminal
  • fabgl::PS2Controller with KbdMode::NoVirtualKeys: the keyboard driver delivers scancodes without translating them
  • Keyboard low-level commands: identification, reset, scancode set selection and LED control
  • fabgl::UARTStream to read commands from the serial console

Hardware

  • VGA output (default pins)
  • PS/2 keyboard on port 0 (CLK = GPIO33, DATA = GPIO32)
  • USB serial connection (115200 baud) to send commands

Usage

Open a serial monitor on the board port (for example idf.py -p PORT monitor) and type one of these single-character commands (no Enter needed):

Key Command
h Print the help, chip info and keyboard identification
q Select scancode set 1
w Select scancode set 2
l Test the keyboard LEDs (cycles through all 8 combinations, one per second, then turns them off)
r Reset the keyboard and print its identification

Then press keys on the PS/2 keyboard: the scancodes appear on screen and on the serial monitor. For example pressing and releasing A in set 2 prints 1C and F0 1C.

Build

See Building the examples.

87void printInfo()
88{
89 auto keyboard = PS2Controller.keyboard();
90
91 if (keyboard->isKeyboardAvailable()) {
92 xprintf("Device Id = ");
93 switch (keyboard->identify()) {
94 case PS2DeviceType::OldATKeyboard:
95 xprintf("\"Old AT Keyboard\"");
96 break;
97 case PS2DeviceType::MouseStandard:
98 xprintf("\"Standard Mouse\"");
99 break;
100 case PS2DeviceType::MouseWithScrollWheel:
101 xprintf("\"Mouse with scroll wheel\"");
102 break;
103 case PS2DeviceType::Mouse5Buttons:
104 xprintf("\"5 Buttons Mouse\"");
105 break;
106 case PS2DeviceType::MF2KeyboardWithTranslation:
107 xprintf("\"MF2 Keyboard with translation\"");
108 break;
109 case PS2DeviceType::M2Keyboard:
110 xprintf("\"MF2 keyboard\"");
111 break;
112 default:
113 xprintf("\"Unknown\"");
114 break;
115 }
116 xprintf("\r\n", keyboard->getLayout()->name);
117 } else
118 xprintf("Keyboard Error!\r\n");
119}

123void setup()

144void loop()
145{
146 static int clen = 1;
147 auto keyboard = PS2Controller.keyboard();
148
149 if (Console.available() > 0) {
150 char c = Console.read();
151 switch (c) {
152 case 'h':
153 printHelp();
154 break;
155 case 'r':
156 keyboard->reset();
157 printInfo();
158 break;
159 case 'l':
160 for (int i = 0; i < 8; ++i) {
161 keyboard->setLEDs(i & 1, i & 2, i & 4);
162 vTaskDelay(pdMS_TO_TICKS(1000));
163 }
164 vTaskDelay(pdMS_TO_TICKS(2000));
165 if (keyboard->setLEDs(0, 0, 0))
166 xprintf("OK\r\n");
167 break;
168 case 'q':
169 keyboard->setScancodeSet(1);
170 xprintf("Scancode Set = %d\r\n", keyboard->scancodeSet());
171 break;
172 case 'w':
173 keyboard->setScancodeSet(2);
174 xprintf("Scancode Set = %d\r\n", keyboard->scancodeSet());
175 break;
176 }
177 }
178
179 if (keyboard->scancodeAvailable()) {
180 int scode = keyboard->getNextScancode();
181 xprintf("%02X ", scode);
182 if (scode == 0xF0 || scode == 0xE0) ++clen;
183 --clen;
184 if (clen == 0) {
185 clen = 1;
186 xprintf("\r\n");
187 }
188 }
189
190}
