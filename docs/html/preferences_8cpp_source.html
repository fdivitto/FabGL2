51bool Preferences::begin(char const * name, bool readOnly)
52{
53 end();
54
55 auto r = nvs_flash_init();
56 if (r == ESP_ERR_NVS_NO_FREE_PAGES || r == ESP_ERR_NVS_NEW_VERSION_FOUND) {
57 nvs_flash_erase();
58 r = nvs_flash_init();
59 }
60 if (r != ESP_OK)
61 return false;
62
63 m_readOnly = readOnly;
64 m_opened = (nvs_open(name, readOnly ? NVS_READONLY : NVS_READWRITE, &m_handle) == ESP_OK);
65 return m_opened;
66}

69void Preferences::end()
