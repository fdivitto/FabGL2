Songs

Plays simple songs written as text strings, using a small ADSR "sound chip" built on top of fabgl::SoundGenerator. By Carles Oriol.

The program plays, in an endless loop:

  1. a melody with a triangle wave
  2. the same melody with a second, lower sine voice
  3. a three-track tune (sine lead with glides, square bass and noise percussion)

The screen only shows a title; there are no controls.

Song notation

Notes are written as <note>[#|b]<octave> <duration> separated by spaces:

  • note A...G, optional # (sharp) or b (flat), octave 2...7 (for example C#5); P is a pause
  • in play_song1() the duration is a fraction of a second (4 = 1/4 s, 2 = 1/2 s, 16 = 1/16 s)
  • in play_song2() the duration is in beats (1 = 250 ms, 0.5, 1.5, 4, ...)

Example: "A4 4 A4 4 A#4 4 C5 4 P 8". Edit the strings in Songs.cpp to play your own tunes.

What it demonstrates

  • Creating and destroying waveform generators on the fly, one per note, each in its own task
  • Software ADSR envelope (attack, decay, sustain, release) and frequency glides (portamento)
  • Several voices playing at the same time through SoundGenerator

See also SoundChipSimulator (same engine, sound effects) and Synth.

Hardware

  • VGA output (default pins)
  • Audio output on GPIO25

69void iPlaySound( void *pvParameters )
70{
71 playsounddata psd = *(playsounddata *)pvParameters;
72
73 WaveformGenerator *pwave = nullptr;
74 if ( psd.wave == WAVE_SQUARE) pwave = new SquareWaveformGenerator();
75 if ( psd.wave == WAVE_SINE) pwave = new SineWaveformGenerator();
76 if ( psd.wave == WAVE_TRIANGLE) pwave = new TriangleWaveformGenerator();
77 if ( psd.wave == WAVE_SAW) pwave = new SawtoothWaveformGenerator();
78 if ( psd.wave == WAVE_NOISE) pwave = new NoiseWaveformGenerator();
79
80 int sustainVolume = psd.sustain * psd.volume / 127;
81
82 soundGenerator.attach( pwave);
83 pwave->setVolume( ((psd.attack == 0) ? ( (psd.decay != 0) ? psd.volume : sustainVolume) : 0) );
84 pwave->setFrequency( psd.freq_start );
85 pwave->enable(true );
86
87 long startTime = (uint32_t)(esp_timer_get_time() / 1000);
88
89 while ( (uint32_t)(esp_timer_get_time() / 1000) - startTime < psd.durationms)
90 {
91 long current = (uint32_t)(esp_timer_get_time() / 1000) - startTime;
92
93 if ( current < psd.attack ) // ATTACK VOLUME
94 pwave->setVolume( (psd.volume * current) / psd.attack );
95 else if ( current > psd.attack && current < (psd.attack + psd.decay)) // DECAY VOLUME
96 pwave->setVolume( mapValue( current - psd.attack, 0, psd.decay, psd.volume, sustainVolume ) );
97 else if ( current > psd.durationms - psd.release) // RELEASE VOLUME
98 pwave->setVolume( mapValue( current - (psd.durationms - psd.release), 0, psd.release, sustainVolume, 0 ) );
99 else
100 pwave->setVolume( sustainVolume );
101
102
103 if ( psd.modfreq != MODFREQ_NONE)
104 {
105 int maxtime = psd.durationms;
106 if (psd.modfreq == MODFREQ_TO_RELEASE)
107 maxtime = (psd.durationms - psd.release); // until release
108 else if (psd.modfreq == MODFREQ_TO_SUSTAIN)
109 maxtime = (psd.attack + psd.decay); // until sustain
110
111 int f = ((current > maxtime) ? psd.freq_end : (mapValue(current, 0, maxtime, psd.freq_start, psd.freq_end)));
112 pwave->setFrequency ( f ) ;
113 }
114 vTaskDelay(5);
115 }
116
117 soundGenerator.detach( pwave);
118 pwave->enable( false );
119 delete pwave;
120
121 vTaskDelete( NULL );
122}

144// note (C,D,E,F,G,A,B) + [#,b] + octave (2..7) + space + tempo (99..1)
145// pause (P) + space + tempo (99.1)
146char const * noteToFreq(char const * note, int * freq)
147{
148 uint16_t NIDX2FREQ[][12] = { { 66, 70, 74, 78, 83, 88, 93, 98, 104, 110, 117, 124 }, // 2
149 { 131, 139, 147, 156, 165, 175, 185, 196, 208, 220, 233, 247 }, // 3
150 { 262, 277, 294, 311, 330, 349, 370, 392, 415, 440, 466, 494 }, // 4
151 { 523, 554, 587, 622, 659, 698, 740, 784, 831, 880, 932, 988 }, // 5
152 { 1046, 1109, 1175, 1245, 1319, 1397, 1480, 1568, 1661, 1760, 1865, 1976 }, // 6
153 { 2093, 2217, 2349, 2489, 2637, 2794, 2960, 3136, 3322, 3520, 3729, 3951 }, // 7
154 };
155 uint8_t NNAME2NIDX[] = {9, 11, 0, 2, 4, 5, 7}; // A, B, C, D, E, F, G
156 *freq = 0;
157 while (*note && *note == ' ')
158 ++note;
159 if (*note == 0)
160 return note;
161 int noteIndex = (*note >= 'A' && *note <= 'G' ? NNAME2NIDX[*note - 'A'] : -2); // -2 = pause
162 ++note;
163 if (*note == '#') {
164 ++noteIndex;
165 ++note;
166 } else if (*note == 'b') {
167 --noteIndex;
168 ++note;
169 }
170 int octave = *note - '0';
171 ++note;
172 if (noteIndex == -1) {
173 noteIndex = 11;
174 --octave;
175 } else if (noteIndex == 12) {
176 noteIndex = 0;
177 ++octave;
178 }
179 if (noteIndex >= 0 && noteIndex <= 11 && octave >= 2 && octave <= 7)
180 *freq = NIDX2FREQ[octave - 2][noteIndex];
181 return note;
182}

185char const * noteToDelay(char const * note, int * delayMS)
186{
187 *delayMS = 0;
188 while (*note && *note == ' ')
189 ++note;
190 if (*note == 0)
191 return note;
192 int val = atoi(note);
193 if (val > 0)
194 *delayMS = 1000 / val;
195 return note + (val > 9 ? 2 : 1);
196}
197
198// 0,5 = cortxea, 1 = negra 2 = blanca...
199char const * noteToDelayEx(char const * note, int * delayMS)
200{
201 *delayMS = 0;
202 while (*note && *note == ' ')
203 ++note;
204 if (*note == 0)
205 return note;
206 double val = atof(note);
207 if (val > 0)
208 *delayMS = 250 * val;
209
210 while ((*note && isdigit(*note)) || *note == '.')
211 ++note;
212
213 return note;
214}
