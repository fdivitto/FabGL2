KeyboardVirtualKeys

Diagnostic tool that shows how PS/2 keyboard events are translated into FabGL virtual keys.

For each key press and release the example prints the virtual key name (VK_a, VK_LSHIFT, ...), the resulting ASCII code and character, the direction (DN = pressed, UP = released) and the raw scancodes. It is the quickest way to check a keyboard layout or to find out which VirtualKey a key generates. The output goes both to the VGA screen (640x480, 16 colors) and to the serial console.

What it demonstrates

  • fabgl::PS2Controller with virtual key generation (Keyboard::getNextVirtualKey(), VirtualKeyItem)
  • Run-time selection of the keyboard layout (Keyboard::setLayout())
  • Keyboard identification, reset, scancode set selection and LED control
  • fabgl::VGA16Controller with a fabgl::Terminal

Hardware

  • VGA output (default pins)
  • PS/2 keyboard on port 0 (CLK = GPIO33, DATA = GPIO32)
  • USB serial connection (115200 baud) to send commands

Usage

Open a serial monitor on the board port (for example idf.py -p PORT monitor) and type one of these single-character commands:

Key Command
h Print the help, chip info, keyboard identification and current layout
1 ... 8 Select the layout: 1 US, 2 UK, 3 German, 4 Italian, 5 Spanish, 6 French, 7 Belgian, 8 Norwegian
r Reset the keyboard
q / w Select scancode set 1 / 2
l Test the keyboard LEDs

Then press keys on the PS/2 keyboard. Example of output line:

VK_a: ASCII = 0x61 'a' DN [1C ]

Build

See Building the examples.

92void printInfo()
93{
94 auto keyboard = PS2Controller.keyboard();
95
96 if (keyboard->isKeyboardAvailable()) {
97 xprintf("Device Id = ");
98 switch (keyboard->identify()) {
99 case PS2DeviceType::OldATKeyboard:
100 xprintf("\"Old AT Keyboard\"");
101 break;
102 case PS2DeviceType::MouseStandard:
103 xprintf("\"Standard Mouse\"");
104 break;
105 case PS2DeviceType::MouseWithScrollWheel:
106 xprintf("\"Mouse with scroll wheel\"");
107 break;
108 case PS2DeviceType::Mouse5Buttons:
109 xprintf("\"5 Buttons Mouse\"");
110 break;
111 case PS2DeviceType::MF2KeyboardWithTranslation:
112 xprintf("\"MF2 Keyboard with translation\"");
113 break;
114 case PS2DeviceType::M2Keyboard:
115 xprintf("\"MF2 keyboard\"");
116 break;
117 default:
118 xprintf("\"Unknown\"");
119 break;
120 }
121 xprintf(" Keyboard Layout: \"%s\"\r\n", keyboard->getLayout()->name);
122 } else
123 xprintf("Keyboard Error!\r\n");
124}

151void loop()
152{
153 auto keyboard = PS2Controller.keyboard();
154
155 //static fabgl::VirtualKey lastvk = fabgl::VK_NONE; // avoid to repeat last vk
156
157 if (Console.available() > 0) {
158 char c = Console.read();
159 switch (c) {
160 case 'h':
161 printHelp();
162 break;
163 case '1':
164 keyboard->setLayout(&fabgl::USLayout);
165 printInfo();
166 break;
167 case '2':
168 keyboard->setLayout(&fabgl::UKLayout);
169 printInfo();
170 break;
171 case '3':
172 keyboard->setLayout(&fabgl::GermanLayout);
173 printInfo();
174 break;
175 case '4':
176 keyboard->setLayout(&fabgl::ItalianLayout);
177 printInfo();
178 break;
179 case '5':
180 keyboard->setLayout(&fabgl::SpanishLayout);
181 printInfo();
182 break;
183 case '6':
184 keyboard->setLayout(&fabgl::FrenchLayout);
185 printInfo();
186 break;
187 case '7':
188 keyboard->setLayout(&fabgl::BelgianLayout);
189 printInfo();
190 break;
191 case '8':
192 keyboard->setLayout(&fabgl::NorwegianLayout);
193 printInfo();
194 break;
195 case 'r':
196 keyboard->reset();
197 printInfo();
198 break;
199 case 'l':
200 for (int i = 0; i < 8; ++i) {
201 keyboard->setLEDs(i & 1, i & 2, i & 4);
202 vTaskDelay(pdMS_TO_TICKS(1000));
203 }
204 vTaskDelay(pdMS_TO_TICKS(2000));
205 if (keyboard->setLEDs(0, 0, 0))
206 xprintf("OK\r\n");
207 break;
208 case 'q':
209 keyboard->setScancodeSet(1);
210 xprintf("Scancode Set = %d\r\n", keyboard->scancodeSet());
211 break;
212 case 'w':
213 keyboard->setScancodeSet(2);
214 xprintf("Scancode Set = %d\r\n", keyboard->scancodeSet());
215 break;
216 }
217 }
218
219 if (keyboard->virtualKeyAvailable()) {
220 // ascii mode (show ASCIIl, VirtualKeys and scancodes)
221 VirtualKeyItem item;
222 if (keyboard->getNextVirtualKey(&item)) {
223 xprintf("%s: ", keyboard->virtualKeyToString(item.vk));
224 xprintf("\tASCII = 0x%02X\t", item.ASCII);
225 if (item.ASCII >= ' ')
226 xprintf("'%c'", item.ASCII);
227 xprintf("\t%s", item.down ? "DN" : "UP");
228 xprintf("\t[");
229 for (int i = 0; i < 8 && item.scancode[i] != 0; ++i)
230 xprintf("%02X ", item.scancode[i]);
231 xprintf("]");
232 xprintf("\r\n");
233 }
234 }
235
236}
