NetworkTerminal

A WiFi network terminal with a small command shell: scan and connect to WiFi networks, ping hosts and open telnet sessions to remote computers, BBSes and servers. It runs in the VGA text mode (80x25, 16 colors).

At startup it runs info, keyb us and scan automatically, then shows the > prompt.

Commands

Command Description
help Shows the list of commands
info Shows screen and terminal size, free memory, WiFi SSID and IP address
scan Scans the WiFi networks: SSID, signal strength (dBm), channel and encryption
wifi SSID [PASSWORD] Connects to a WiFi network (omit the password for open networks). Example: wifi MyWifi MyPassword
telnet HOST [PORT] Opens a telnet session with HOST (IP address or host name), default port 23. Example: telnet towel.blinkenlights.nl
ping HOST Pings HOST every second until Ctrl+C is pressed, then prints the statistics. Example: ping 8.8.8.8
keyb LAYOUT Sets the keyboard layout: us, uk, de, it, es, fr, be, no
reboot Restarts the board

The command line supports editing with the arrow keys, Home, End, Backspace and Delete (fabgl::LineEditor).

During a telnet session every key is sent to the remote host. The session ends (and the prompt comes back) when the remote host closes the connection. The terminal reports itself as wsvt25 with a 80x25 window size.

What it demonstrates

  • fabgl::Terminal in local mode + fabgl::LineEditor to build a command line interface
  • fabgl::WiFiScanner, fabgl::WiFiConnection, fabgl::TCPClient (telnet with basic option negotiation), fabgl::ICMP (ping)
  • fabgl::VGATextController: the text mode leaves most of the RAM free for WiFi

Hardware

  • VGA output (default pins)
  • PS/2 keyboard on port 0 (CLK = GPIO33, DATA = GPIO32)

121void decode_command()
122{
123 auto inputLine = LineEditor.get();
124 if (*inputLine == 0)
125 state = State::Prompt;
126 else if (strncmp(inputLine, "help", 4) == 0)
127 state = State::Help;
128 else if (strncmp(inputLine, "info", 4) == 0)
129 state = State::Info;
130 else if (strncmp(inputLine, "wifi", 4) == 0)
131 state = State::Wifi;
132 else if (strncmp(inputLine, "telnet", 6) == 0)
133 state = State::TelnetInit;
134 else if (strncmp(inputLine, "scan", 4) == 0)
135 state = State::Scan;
136 else if (strncmp(inputLine, "ping", 4) == 0)
137 state = State::Ping;
138 else if (strncmp(inputLine, "reboot", 6) == 0)
139 state = State::Reset;
140 else if (strncmp(inputLine, "keyb", 4) == 0)
141 state = State::Keyb;
142 else
143 state = State::UnknownCommand;
144}

147void exe_prompt()
148{
149 if (currentScript) {
150 // process commands from script
151 if (*currentScript == 0 || error) {
152 // end of script, return to prompt
153 currentScript = nullptr;
154 state = State::Prompt;
155 } else {
156 // execute current line and move to the next one
157 int linelen = strchr(currentScript, '\r') - currentScript;
158 LineEditor.setText(currentScript, linelen);
159 currentScript += linelen + 1;
160 decode_command();
161 }
162 } else {
163 // process commands from keyboard
164 Terminal.write(">");
165 state = State::PromptInput;
166 }
167}

178void exe_scan()
179{
180 Terminal.write("Scanning...");
181 Terminal.flush();
182 fabgl::WiFiScanner scanner;
183 int networksCount = scanner.scan(32) ? scanner.count() : 0;
184 Terminal.printf("%d network(s) found\r\n", networksCount);
185 if (networksCount) {
186 Terminal.write ("\e[90m #\e[4GSSID\e[45GRSSI\e[55GCh\e[60GEncryption\e[92m\r\n");
187 for (int i = 0; i < networksCount; ++i)
188 Terminal.printf("\e[93m %d\e[4G%s\e[93m\e[45G%d dBm\e[55G%d\e[60G%s\e[92m\r\n", i + 1, (char const *) scanner.get(i)->ssid, scanner.get(i)->rssi, scanner.get(i)->primary, fabgl::WiFiScanner::authModeStr(scanner.get(i)->authmode));
189 }
190 error = false;
191 state = State::Prompt;
192}

195void exe_wifi()
196{
197 static const int MAX_SSID_SIZE = 32;
198 static const int MAX_PSW_SIZE = 32;
199 char ssid[MAX_SSID_SIZE + 1];
200 char psw[MAX_PSW_SIZE + 1] = {0};
201 error = true;
202 auto inputLine = LineEditor.get();
203 if (sscanf(inputLine, "wifi %32s %32s", ssid, psw) >= 1) {
204 Terminal.write("Connecting WiFi...");
205 Terminal.flush();
206 wifi.disconnect();
207 for (int i = 0; i < 2; ++i) {
208 if (wifi.connect(ssid, psw, 10000) == fabgl::WiFiConnectionState::Connected)
209 break;
210 wifi.disconnect();
211 }
212 if (wifi.state() == fabgl::WiFiConnectionState::Connected) {
213 Terminal.printf("connected to %s, IP is %s\r\n", wifi.SSID(), wifi.IPString());
214 error = false;
215 } else {
216 Terminal.write("failed!\r\n");
217 }
218 }
219 state = State::Prompt;
220}

223void exe_telnetInit()
224{
225 static const int MAX_HOST_SIZE = 32;
226 char host[MAX_HOST_SIZE + 1];
227 int port;
228 error = true;
229 auto inputLine = LineEditor.get();
230 int pCount = sscanf(inputLine, "telnet %32s %d", host, &port);
231 if (pCount > 0) {
232 if (pCount == 1)
233 port = 23;
234 Terminal.printf("Trying %s...\r\n", host);
235 if (client.connect(host, port)) {
236 Terminal.printf("Connected to %s\r\n", host);
237 error = false;
238 state = State::Telnet;
239 } else {
240 Terminal.write("Unable to connect to remote host\r\n");
241 state = State::Prompt;
242 }
243 } else {
244 Terminal.write("Mistake\r\n");
245 state = State::Prompt;
246 }
247}

258void exe_telnet()
259{
260 // process data from remote host (up to 1024 codes at the time)
261 for (int i = 0; client.available() && i < 1024; ++i) {
262 int c = client.read();
263 if (c == 0xFF) {
264 // IAC (Interpret As Command)
265 uint8_t cmd = clientWaitForChar();
266 uint8_t opt = clientWaitForChar();
267 if (cmd == 0xFD && opt == 0x1F) {
268 // DO WINDOWSIZE
269 client.write("\xFF\xFB\x1F", 3); // IAC WILL WINDOWSIZE
270 client.write("\xFF\xFA\x1F" "\x00\x50\x00\x19" "\xFF\xF0", 9); // IAC SB WINDOWSIZE 0 80 0 25 IAC SE
271 } else if (cmd == 0xFD && opt == 0x18) {
272 // DO TERMINALTYPE
273 client.write("\xFF\xFB\x18", 3); // IAC WILL TERMINALTYPE
274 } else if (cmd == 0xFA && opt == 0x18) {
275 // SB TERMINALTYPE
276 c = clientWaitForChar(); // bypass '1'
277 c = clientWaitForChar(); // bypass IAC
278 c = clientWaitForChar(); // bypass SE
279 client.write("\xFF\xFA\x18\x00" "wsvt25" "\xFF\xF0", 12); // IAC SB TERMINALTYPE 0 "...." IAC SE
280 } else {
281 uint8_t pck[3] = {0xFF, 0, opt};
282 if (cmd == 0xFD) // DO -> WONT
283 pck[1] = 0xFC;
284 else if (cmd == 0xFB) // WILL -> DO
285 pck[1] = 0xFD;
286 client.write(pck, 3);
287 }
288 } else {
289 Terminal.write(c);
290 }
291 }
292 // process data from terminal (keyboard)
293 while (Terminal.available()) {
294 client.write( Terminal.read() );
295 }
296 // return to prompt?
297 if (!client.connected()) {
298 client.stop();
299 state = State::Prompt;
300 }
301}

304void exe_ping()
305{
306 char host[64];
307 auto inputLine = LineEditor.get();
308 int pcount = sscanf(inputLine, "ping %s", host);
309 if (pcount > 0) {
310 int sent = 0, recv = 0;
311 fabgl::ICMP icmp;
312 while (true) {
313
314 // CTRL-C ?
315 if (Terminal.available() && Terminal.read() == 0x03)
316 break;
317
318 int t = icmp.ping(host);
319 if (t >= 0) {
320 Terminal.printf("%d bytes from %s: icmp_seq=%d ttl=%d time=%.3f ms\r\n", icmp.receivedBytes(), icmp.hostIPString(), icmp.receivedSeq(), icmp.receivedTTL(), (double)t/1000.0);
321 vTaskDelay(pdMS_TO_TICKS(1000));
322 ++recv;
323 } else if (t == -2) {
324 Terminal.printf("Cannot resolve %s: Unknown host\r\n", host);
325 break;
326 } else {
327 Terminal.printf("Request timeout for icmp_seq %d\r\n", icmp.receivedSeq());
328 }
329 ++sent;
330
331 }
332 if (sent > 0) {
333 Terminal.printf("--- %s ping statistics ---\r\n", host);
334 Terminal.printf("%d packets transmitted, %d packets received, %.1f%% packet loss\r\n", sent, recv, (double)(sent - recv) / sent * 100.0);
335 }
336 }
337 state = State::Prompt;
338}

341void exe_keyb()
342{
343 if (PS2Controller.keyboard()->isKeyboardAvailable()) {
344 char layout[3];
345 auto inputLine = LineEditor.get();
346 if (sscanf(inputLine, "keyb %2s", layout) == 1) {
347 if (strcasecmp(layout, "US") == 0)
348 Terminal.keyboard()->setLayout(&fabgl::USLayout);
349 else if (strcasecmp(layout, "UK") == 0)
350 Terminal.keyboard()->setLayout(&fabgl::UKLayout);
351 else if (strcasecmp(layout, "DE") == 0)
352 Terminal.keyboard()->setLayout(&fabgl::GermanLayout);
353 else if (strcasecmp(layout, "IT") == 0)
354 Terminal.keyboard()->setLayout(&fabgl::ItalianLayout);
355 else if (strcasecmp(layout, "ES") == 0)
356 Terminal.keyboard()->setLayout(&fabgl::SpanishLayout);
357 else if (strcasecmp(layout, "FR") == 0)
358 Terminal.keyboard()->setLayout(&fabgl::FrenchLayout);
359 else if (strcasecmp(layout, "BE") == 0)
360 Terminal.keyboard()->setLayout(&fabgl::BelgianLayout);
361 else if (strcasecmp(layout, "NO") == 0)
362 Terminal.keyboard()->setLayout(&fabgl::NorwegianLayout);
363 else {
364 Terminal.printf("Error! Invalid keyboard layout.\r\n");
365 state = State::Prompt;
366 return;
367 }
368 }
369 Terminal.printf("\r\nKeyboard layout is : \e[93m%s\e[92m\r\n\r\n", Terminal.keyboard()->getLayout()->desc);
370 } else {
371 Terminal.printf("No keyboard present\r\n");
372 }
373 state = State::Prompt;
374}

399void loop()
400{
401 switch (state) {
402
403 case State::Prompt:
404 exe_prompt();
405 break;
406
407 case State::PromptInput:
408 exe_promptInput();
409 break;
410
411 case State::Help:
412 exe_help();
413 break;
414
415 case State::Info:
416 exe_info();
417 break;
418
419 case State::Wifi:
420 exe_wifi();
421 break;
422
423 case State::TelnetInit:
424 exe_telnetInit();
425 break;
426
427 case State::Telnet:
428 exe_telnet();
429 break;
430
431 case State::Scan:
432 exe_scan();
433 break;
434
435 case State::Ping:
436 exe_ping();
437 break;
438
439 case State::Reset:
440 esp_restart();
441 break;
442
443 case State::Keyb:
444 exe_keyb();
445 break;
446
447 case State::UnknownCommand:
448 Terminal.write("\r\nMistake\r\n");
449 state = State::Prompt;
450 break;
451
452 default:
453 Terminal.write("\r\nNot Implemeted\r\n");
454 state = State::Prompt;
455 break;
456
457 }
458}
