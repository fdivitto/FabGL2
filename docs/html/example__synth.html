Synth

FabGL Synth: an 8-voice polyphonic synthesizer played with the PS/2 keyboard (like a piano) or with the mouse on an on-screen keyboard. Each voice has its own ADSR envelope and resonant low-pass filter, all computed in fixed point inside the sound generator interrupt.

Screen

  • Waveform: Sine, Square, Pulse 25%, Triangle, Sawtooth
  • Octave: Z < and > X buttons move the keyboard down and up one octave (from C1 to C6)
  • Volume: master volume
  • Envelope (ADSR): Attack (0-1000 ms), Decay (0-1000 ms), Sustain (0-100 %), Release (0-2000 ms)
  • Low-pass filter: Cutoff (50-7000 Hz, logarithmic), Resonance (Q from 0.7, no resonance, upwards), On/Off
  • Now playing: name and frequency of the last note
  • On-screen keyboard: two octaves plus the final C. Keys pressed from the PS/2 keyboard and from the mouse are highlighted; the letters of the PS/2 keys are drawn on the keys

All the controls change the sound while you play.

Keys

The PS/2 keyboard works like a piano keyboard (two rows of keys, the upper row gives the black keys):

W E T Y U O P ] black keys: C# D# F# G# A# C# D# F#
A S D F G H J K L ; ' white keys: C D E F G A B C D E F
Key Action
A W S E D F T G Y H U J K O L P ; ' ] Notes, from C to F# of the next octave (on the Italian layout ò, à, + replace ;, ', ])
Z / X Octave down / up
1 ... 5 Select the waveform

With the mouse, click a key of the on-screen keyboard; drag with the button pressed to slide between notes (glissando).

When more than 8 notes are requested, the synth reuses a voice that is releasing, otherwise the oldest one.

What it demonstrates

  • Custom WaveformGenerator subclasses (the voices) attached to fabgl::SoundGenerator
  • Real-time DSP inside the sound generator interrupt: ADSR envelope and RBJ biquad low-pass filter in Q28 fixed point (the FPU cannot be used in the interrupt); coefficients are computed by the UI task and swapped atomically
  • Key down / key up handling with virtual keys, ignoring typematic repeats
  • A custom-painted GUI control (the piano keyboard) with mouse hit testing

Hardware

  • VGA output (default pins)
  • Audio output on GPIO25
  • PS/2 keyboard on port 0 and PS/2 mouse on port 1

67static constexpr int ENV_MAX = 1 << 24; // envelope full level (fixed point)
68
69
70struct Envelope {
71 volatile int attackInc = ENV_MAX;
72 volatile int decayInc = ENV_MAX;
73 volatile int sustain = ENV_MAX;
74 volatile int releaseInc = ENV_MAX;
75
76 // ms -> level increment per sample (from 0 to ENV_MAX)
77 static int msToInc(int ms) {
78 int samples = ms * FABGL_SOUNDGEN_DEFAULT_SAMPLE_RATE / 1000;
79 return samples > 0 ? ENV_MAX / samples : ENV_MAX;
80 }
81
82 void setAttack(int ms) { attackInc = msToInc(ms); }
83 void setDecay(int ms) { decayInc = msToInc(ms); }
84 void setSustain(int percent) { sustain = (int64_t) ENV_MAX * percent / 100; }
85 void setRelease(int ms) { releaseInc = msToInc(ms); }
86};

93// Resonant low-pass filter parameters (shared by all voices)
94//
95// Second order (12 dB/octave) "RBJ cookbook" biquad. Coefficients are computed in floating point by
96// the UI task and used in fixed point (Q28) by the sound generator interrupt, where the FPU can't be used.
97// Two coefficient sets are kept: the UI writes the inactive one and then swaps them, so the interrupt
98// never sees a half-updated set.
99
100
101struct LowPassFilter {
102
103 static constexpr int COEFF_SHIFT = 28;
104
105 static constexpr int MIN_CUTOFF = 50; // Hz
106 static constexpr int MAX_CUTOFF = 7000; // Hz (sample rate is 16384 Hz, Nyquist is 8192 Hz)
107 static constexpr float MIN_Q = 0.707f; // no resonance (Butterworth)
108 static constexpr float MAX_Q = 12.0f;
109
110 struct Coeffs {
111 int32_t b0, b1, b2, a1, a2;
112 };
113
114 Coeffs sets[2];
115 volatile int active = 0;
116 volatile bool enabled = true;
117 int cutoff = 2000; // Hz
118 float q = 2.0f;
119
120 LowPassFilter() { update(); }
121
122 void setCutoff(int value) { cutoff = value; update(); }
123 void setResonance(float value) { q = value; update(); }
124
125 void update()
126 {
127 float w0 = 2.0f * (float) M_PI * cutoff / FABGL_SOUNDGEN_DEFAULT_SAMPLE_RATE;
128 float cosw = cosf(w0);
129 float alpha = sinf(w0) / (2.0f * q);
130 float a0 = 1.0f + alpha;
131 float scale = (float) (1 << COEFF_SHIFT) / a0;
132
133 int next = active ^ 1;
134 Coeffs & c = sets[next];
135 c.b0 = (int32_t) lroundf((1.0f - cosw) * 0.5f * scale);
136 c.b1 = (int32_t) lroundf((1.0f - cosw) * scale);
137 c.b2 = c.b0;
138 c.a1 = (int32_t) lroundf(-2.0f * cosw * scale);
139 c.a2 = (int32_t) lroundf((1.0f - alpha) * scale);
140 active = next;
141 }
142
143 // slider position (0..100) <-> logarithmic scales
144 static int cutoffFromSlider(int pos) { return (int) lroundf(MIN_CUTOFF * powf((float) MAX_CUTOFF / MIN_CUTOFF, pos / 100.0f)); }
145 static float qFromSlider(int pos) { return MIN_Q * powf(MAX_Q / MIN_Q, pos / 100.0f); }
146};

153// Voice: a waveform generator (sine, square, ...) shaped by the ADSR envelope
154
155
156enum class Waveform { Sine, Square, Pulse, Triangle, Sawtooth };
157
158static const char * WaveformNames[] = { "Sine", "Square", "Pulse 25%", "Triangle", "Sawtooth" };
159static constexpr int WaveformsCount = sizeof(WaveformNames) / sizeof(WaveformNames[0]);
160
161
162class Voice : public WaveformGenerator {
163
164public:
165
166 enum class State { Idle, Attack, Decay, Sustain, Release };
167
168 Voice()
169 {
170 m_pulse.setDutyCycle(64);
171 m_gen = &m_sine;
172 }
173
174 // must be called after attaching the voice to the sound generator (sample rate is known)
175 void setWaveform(Waveform waveform)
176 {
177 WaveformGenerator * gens[] = { &m_sine, &m_square, &m_pulse, &m_triangle, &m_sawtooth };
178 auto gen = gens[(int)waveform];
179 gen->setSampleRate(sampleRate());
180 gen->setVolume(127);
181 gen->setFrequency(m_frequency);
182 m_gen = gen;
183 }
184
185 void setFrequency(int value)
186 {
187 m_frequency = value;
188 m_gen->setFrequency(value);
189 }
190
191 void noteOn(int note, int frequency)
192 {
193 m_note = note;
194 setFrequency(frequency);
195 if (m_state == State::Idle)
196 m_x1 = m_x2 = m_y1 = m_y2 = 0; // clear filter state (safe: the interrupt doesn't touch idle voices)
197 m_state = State::Attack; // a retriggered voice restarts from its current level (no clicks)
198 enable(true);
199 }
200
201 void noteOff()
202 {
203 if (m_state != State::Idle)
204 m_state = State::Release;
205 }
206
207 int note() { return m_note; }
208 State state() { return m_state; }
209 bool isIdle() { return m_state == State::Idle; }
210
211 int getSample()
212 {
213 switch (m_state) {
214 case State::Attack:
215 m_level += envelope.attackInc;
216 if (m_level >= ENV_MAX) {
217 m_level = ENV_MAX;
218 m_state = State::Decay;
219 }
220 break;
221 case State::Decay:
222 m_level -= envelope.decayInc;
223 if (m_level <= envelope.sustain) {
224 m_level = envelope.sustain;
225 m_state = State::Sustain;
226 }
227 break;
228 case State::Sustain:
229 m_level = envelope.sustain; // follows the sustain slider while the key is held
230 break;
231 case State::Release:
232 m_level -= envelope.releaseInc;
233 if (m_level <= 0) {
234 m_level = 0;
235 m_state = State::Idle;
236 m_note = -1;
237 enable(false);
238 }
239 break;
240 case State::Idle:
241 return 0;
242 }
243
244 int sample = m_gen->getSample(); // -128..127 (generator volume = 127)
245 if (lowPassFilter.enabled)
246 sample = filter(sample);
247 sample = sample * volume() / 127; // voice volume
248 return (sample * (m_level >> 16)) >> 8; // envelope (0..256)
249 }
250
251private:
252
253 // biquad low-pass filter, direct form I. Samples are scaled by 256 to keep precision.
254 int filter(int in)
255 {
256 auto const & c = lowPassFilter.sets[lowPassFilter.active];
257 int32_t x0 = in << 8;
258 int64_t acc = (int64_t) c.b0 * x0 + (int64_t) c.b1 * m_x1 + (int64_t) c.b2 * m_x2
259 - (int64_t) c.a1 * m_y1 - (int64_t) c.a2 * m_y2;
260 int32_t y0 = (int32_t) (acc >> LowPassFilter::COEFF_SHIFT);
261 y0 = fabgl::iclamp(y0, -(1 << 20), (1 << 20)); // safety net against runaway (never reached when stable)
262 m_x2 = m_x1;
263 m_x1 = x0;
264 m_y2 = m_y1;
265 m_y1 = y0;
266 return fabgl::iclamp(y0 >> 8, -128, 127); // resonance can exceed the 8 bit range: clip
267 }
268
269 SineWaveformGenerator m_sine;
270 SquareWaveformGenerator m_square;
271 SquareWaveformGenerator m_pulse;
272 TriangleWaveformGenerator m_triangle;
273 SawtoothWaveformGenerator m_sawtooth;
274
275 WaveformGenerator * m_gen;
276 int m_frequency = 0;
277 int m_note = -1;
278 volatile State m_state = State::Idle;
279 int m_level = 0;
280
281 // filter state
282 int32_t m_x1 = 0, m_x2 = 0, m_y1 = 0, m_y2 = 0;
283};

288// Synth: a pool of voices
289
290
291class Synth {
292
293public:
294
295 static constexpr int MAX_VOICES = 8;
296
297 void begin()
298 {
299 for (auto & v : m_voices) {
300 soundGenerator.attach(&v);
301 v.setWaveform(m_waveform);
302 }
303 soundGenerator.play(true);
304 }
305
306 void setWaveform(Waveform value)
307 {
308 m_waveform = value;
309 bool playing = soundGenerator.play(false);
310 for (auto & v : m_voices)
311 v.setWaveform(value);
312 soundGenerator.play(playing);
313 }
314
315 Waveform waveform() { return m_waveform; }
316
317 // note: MIDI note number (60 = C4, 69 = A4 = 440 Hz)
318 void noteOn(int note)
319 {
320 Voice * voice = nullptr;
321 // same note still sounding (releasing)? reuse its voice
322 for (auto & v : m_voices)
323 if (v.note() == note) {
324 voice = &v;
325 break;
326 }
327 // free voice?
328 if (!voice)
329 for (auto & v : m_voices)
330 if (v.isIdle()) {
331 voice = &v;
332 break;
333 }
334 // all voices busy: steal a releasing one, otherwise the oldest one
335 if (!voice)
336 for (auto & v : m_voices)
337 if (v.state() == Voice::State::Release) {
338 voice = &v;
339 break;
340 }
341 if (!voice)
342 voice = &m_voices[m_nextSteal++ % MAX_VOICES];
343
344 voice->noteOn(note, noteFrequency(note));
345 }
346
347 void noteOff(int note)
348 {
349 for (auto & v : m_voices)
350 if (v.note() == note && v.state() != Voice::State::Release)
351 v.noteOff();
352 }
353
354 static int noteFrequency(int note)
355 {
356 return (int) lroundf(440.0f * powf(2.0f, (note - 69) / 12.0f));
357 }
358
359private:
360
361 Voice m_voices[MAX_VOICES];
362 Waveform m_waveform = Waveform::Sawtooth;
363 int m_nextSteal = 0;
364};

385// PianoKeyboard: on screen keyboard control (two octaves plus the final C)

387
388class PianoKeyboard : public uiControl {
389
390public:
391
392 static constexpr int KEYS = 25; // semitones displayed (C..C'')
393 static constexpr int WHITE_KEYS = 15;
394
395 // keyboard source flags
396 static constexpr uint8_t SRC_KEYBOARD = 1;
397 static constexpr uint8_t SRC_MOUSE = 2;
398
399 PianoKeyboard(uiWindow * parent, const Point & pos, const Size & size)
400 : uiControl(parent, pos, size, true)
401 {
402 windowStyle().borderSize = 0;
403 windowProps().focusable = false; // keyboard events are handled by the main frame
404 memset(m_pressed, 0, sizeof(m_pressed));
405 }
406
407 // base note (the leftmost C)
408 void setBaseNote(int value)
409 {
410 releaseAll();
411 m_baseNote = value;
412 repaint();
413 }
414
415 int baseNote() { return m_baseNote; }
416
417 // index: 0..KEYS-1
418 void press(int index, uint8_t source)
419 {
420 if (index < 0 || index >= KEYS || (m_pressed[index] & source))
421 return; // ignore typematic repeats
422 bool wasPressed = m_pressed[index];
423 m_pressed[index] |= source;
424 if (!wasPressed) {
425 synth.noteOn(m_baseNote + index);
426 repaintKey(index);
427 onNote(m_baseNote + index);
428 }
429 }
430
431 void release(int index, uint8_t source)
432 {
433 if (index < 0 || index >= KEYS || !(m_pressed[index] & source))
434 return;
435 m_pressed[index] &= ~source;
436 if (!m_pressed[index]) {
437 synth.noteOff(m_baseNote + index);
438 repaintKey(index);
439 }
440 }
441
442 void releaseAll()
443 {
444 for (int i = 0; i < KEYS; ++i)
445 if (m_pressed[i]) {
446 m_pressed[i] = 0;
447 synth.noteOff(m_baseNote + i);
448 }
449 m_mouseKey = -1;
450 repaint();
451 }
452
453 // keyboard letters to show on the keys (nullptr = none)
454 void setKeyLabels(char const * const * labels) { m_keyLabels = labels; }
455
456 // called when a note starts playing
457 Delegate<int> onNote;
458
459 virtual void processEvent(uiEvent * event)
460 {
461 uiControl::processEvent(event);
462
463 switch (event->id) {
464
465 case fabgl::UIEVT_PAINT:
466 beginPaint(event, uiControl::clientRect(uiOrigin::Window));
467 paintKeyboard();
468 break;
469
470 case fabgl::UIEVT_MOUSEBUTTONDOWN:
471 if (event->params.mouse.changedButton == 1) {
472 m_mouseKey = keyAt(event->params.mouse.status.X, event->params.mouse.status.Y);
473 press(m_mouseKey, SRC_MOUSE);
474 }
475 break;
476
477 case fabgl::UIEVT_MOUSEMOVE:
478 // glissando: dragging the mouse with the button pressed changes note
479 if (m_mouseKey >= 0 && app()->capturedMouseWindow() == this) {
480 int key = keyAt(event->params.mouse.status.X, event->params.mouse.status.Y);
481 if (key != m_mouseKey) {
482 release(m_mouseKey, SRC_MOUSE);
483 m_mouseKey = key;
484 press(m_mouseKey, SRC_MOUSE);
485 }
486 }
487 break;
488
489 case fabgl::UIEVT_MOUSEBUTTONUP:
490 if (event->params.mouse.changedButton == 1) {
491 release(m_mouseKey, SRC_MOUSE);
492 m_mouseKey = -1;
493 }
494 break;
495
496 default:
497 break;
498 }
499 }
500
501private:
502
503 static bool isBlack(int index)
504 {
505 static const bool black[12] = { false, true, false, true, false, false, true, false, true, false, true, false };
506 return black[index % 12];
507 }
508
509 int whiteWidth() { return clientSize().width / WHITE_KEYS; }
510 int blackWidth() { return whiteWidth() * 2 / 3; }
511 int blackHeight() { return clientSize().height * 6 / 10; }
512
513 // number of white keys before "index"
514 static int whiteIndex(int index)
515 {
516 static const int whites[12] = { 0, 1, 1, 2, 2, 3, 4, 4, 5, 5, 6, 6 };
517 return index / 12 * 7 + whites[index % 12];
518 }
519
520 Rect keyRect(int index)
521 {
522 int ww = whiteWidth();
523 if (isBlack(index)) {
524 int x = whiteIndex(index) * ww - blackWidth() / 2;
525 return Rect(x, 0, x + blackWidth() - 1, blackHeight() - 1);
526 } else {
527 int x = whiteIndex(index) * ww;
528 return Rect(x, 0, x + ww - 1, clientSize().height - 1);
529 }
530 }
531
532 int keyAt(int x, int y)
533 {
534 // black keys are over white keys: check them first
535 for (int i = 0; i < KEYS; ++i)
536 if (isBlack(i) && keyRect(i).contains(x, y))
537 return i;
538 for (int i = 0; i < KEYS; ++i)
539 if (!isBlack(i) && keyRect(i).contains(x, y))
540 return i;
541 return -1;
542 }
543
544 void repaintKey(int index)
545 {
546 // a white key is partially covered by black keys: repaint the whole keyboard area of the key
547 repaint(keyRect(index));
548 }
549
550 void paintKey(int index)
551 {
552 Rect r = keyRect(index);
553 bool black = isBlack(index);
554 bool pressed = m_pressed[index] != 0;
555
556 if (black) {
557 canvas()->setBrushColor(pressed ? RGB888(0, 0, 255) : RGB888(0, 0, 0));
558 canvas()->fillRectangle(r);
559 canvas()->setPenColor(RGB888(64, 64, 64));
560 canvas()->drawRectangle(r);
561 } else {
562 canvas()->setBrushColor(pressed ? RGB888(0, 255, 255) : RGB888(255, 255, 255));
563 canvas()->fillRectangle(r);
564 canvas()->setPenColor(RGB888(0, 0, 0));
565 canvas()->drawRectangle(r);
566 }
567
568 auto font = &fabgl::FONT_std_12;
569 canvas()->setGlyphOptions(GlyphOptions().FillBackground(false));
570
571 // keyboard letter
572 if (m_keyLabels && m_keyLabels[index]) {
573 char const * lbl = m_keyLabels[index];
574 int tx = r.X1 + (r.width() - canvas()->textExtent(font, lbl)) / 2;
575 canvas()->setPenColor(black ? RGB888(255, 255, 255) : RGB888(0, 0, 255));
576 canvas()->drawText(font, tx, r.Y2 - font->height - (black ? 4 : 24), lbl);
577 }
578
579 // note name on white keys
580 if (!black) {
581 char name[16];
582 noteName(m_baseNote + index, name, sizeof(name));
583 int tx = r.X1 + (r.width() - canvas()->textExtent(font, name)) / 2;
584 canvas()->setPenColor(RGB888(128, 128, 128));
585 canvas()->drawText(font, tx, r.Y2 - font->height - 4, name);
586 }
587 }
588
589 void paintKeyboard()
590 {
591 // background (area not covered by keys)
592 canvas()->setBrushColor(RGB888(64, 64, 64));
593 canvas()->fillRectangle(uiControl::clientRect(uiOrigin::Window));
594 // white keys first, then black keys on top
595 for (int i = 0; i < KEYS; ++i)
596 if (!isBlack(i))
597 paintKey(i);
598 for (int i = 0; i < KEYS; ++i)
599 if (isBlack(i))
600 paintKey(i);
601 }
602
603 int m_baseNote = 60; // C4
604 uint8_t m_pressed[KEYS]; // SRC_xxx flags
605 int m_mouseKey = -1;
606 char const * const * m_keyLabels = nullptr;
607};
