HardwareTest

Interactive board test with a graphical user interface, useful to check a newly assembled board (for example the TTGO VGA32 or the Olimex ESP32-SBC-FabGL) piece by piece.

The window ("Hardware test", 640x480 VESA mode, 16 colors) contains one row per test:

Test How to use it Expected result
Colors Look at the 16 color boxes 8 normal and 8 bright colors, all distinct. A missing color usually means a broken resistor or VGA pin
Keyboard Click the text box and type Typed characters appear
Sound Press Start (it becomes Stop) Random sine tones every 250 ms on the audio output (GPIO25)
SD card Press Start Mounts the SD card, writes and reads back a 16 KB temporary file: Success!, or Mount Failed! / Write Failed! / Read Failed!
GPIO Choose an input GPIO (In) and an output GPIO (Out), click the output State button The input State button shows the input level (0/1) in real time; the output GPIO follows its State button. Connect Out to In with a wire to test both
Ext GPIO Shown only when an MCP23S17 expander is detected. Click A0...B2 to toggle the outputs, press Play for a LED animation B3...B7 are inputs: they turn bright red when high; INT flashes yellow when the expander raises its interrupt line
WiFi Press Start Scans the WiFi networks: Ok, Found N Networks

The GPIO lists contain only the pins not used by VGA, PS/2 and audio (inputs: 0, 1, 2, 3, 12, 13, 14, 16, 17, 18, 34, 35, 36, 39; outputs: 0, 1, 2, 3, 12, 13, 14, 16, 17, 18). GPIO 1 and 3 are the serial console pins.

What it demonstrates

  • A complete uiApp with frame, labels, color boxes, text edit, buttons, switch buttons, combo boxes and timers
  • fabgl::SoundGenerator, fabgl::FileBrowser (SD card mount), fabgl::WiFiScanner, fabgl::MCP23S17
  • Reducing DisplayController.queueSize to leave enough RAM for both WiFi and the SD card file system

Hardware

  • VGA output (default pins), audio on GPIO25
  • PS/2 keyboard on port 0 and PS/2 mouse on port 1
  • Optional: SD card (TTGO VGA32 slot), MCP23S17 GPIO expander (MISO = 35, or 2 on TTGO VGA32; MOSI = 12; CLK = 14; INT = 36)

275 void updateGPIOInState() {
276 // read internal GPIOs
277 if (gpioInPrevState != gpio_get_level((gpio_num_t)(gpioIn))) {
278 gpioInPrevState = !gpioInPrevState;
279 gpioInState->setText(gpioInPrevState ? "1" : "0");
280 gpioInState->repaint();
281 }
282 // read external GPIOs
283 if (mcp.available()) {
284 // check MCP_INT has been triggered correctly
285 if (gpio_get_level((gpio_num_t)(MCP_INT)) == 0) {
286 extIntLabel->labelStyle().backgroundColor = RGB888(255, 255, 0);
287 extIntLabel->repaint();
288 intOffDelay = 2;
289 } else if (--intOffDelay == 0) {
290 extIntLabel->labelStyle().backgroundColor = RGB888(64, 64, 0);
291 extIntLabel->repaint();
292 }
293 if (mcp.getPortIntFlags(MCP_PORTB)) {
294 // read B3...B7 GPIOs
295 for (int i = MCP_B3; i <= MCP_B7; ++i) {
296 extgpioLabel[i]->labelStyle().backgroundColor = mcp.readGPIO(i) ? RGB888(255, 0, 0) : RGB888(64, 0, 0);
297 extgpioLabel[i]->repaint();
298 }
299 }
300 }

313 void playExtGPIOS() {
314 showWindow(extgpioPlay, false);
315 // single left<->right
316 for (int j = 0; j < 4; ++j) {
317 for (int i = MCP_A0; i <= MCP_B2; ++i) {
318 extGPIOSet(i, true);
319 if (i > MCP_A0)
320 extGPIOSet(i - 1, false);
321 processEvents();
322 vTaskDelay(pdMS_TO_TICKS(80));
323 }
324 for (int i = MCP_B2; i >= MCP_A0; --i) {
325 extGPIOSet(i, true);
326 if (i < MCP_B2)
327 extGPIOSet(i + 1, false);
328 processEvents();
329 vTaskDelay(pdMS_TO_TICKS(80));
330 }
331 }
332 // all together
333 for (int j = 0; j < 4; ++j) {
334 for (int i = MCP_A0; i <= MCP_B2; ++i) {
335 extGPIOSet(i, true);
336 processEvents();
337 vTaskDelay(pdMS_TO_TICKS(80));
338 }
339 for (int i = MCP_A0; i <= MCP_B2; ++i) {
340 extGPIOSet(i, false);
341 processEvents();
342 vTaskDelay(pdMS_TO_TICKS(80));
343 }
344 }
345 // flashing
346 for (int j = 0; j < 10; ++j) {
347 for (int i = MCP_A0; i <= MCP_B2; ++i)
348 extGPIOSet(i, true);
349 processEvents();
350 vTaskDelay(pdMS_TO_TICKS(120));
351 for (int i = MCP_A0; i <= MCP_B2; ++i)
352 extGPIOSet(i, false);
353 processEvents();
354 vTaskDelay(pdMS_TO_TICKS(120));
355 }
356 showWindow(extgpioPlay, true);
357 }

359 void testSD() {
360 // disable MCP (just because SD should be initialized before MCP)
361 bool mcpAvailable = mcp.available();
362 mcp.end();
363 // mount test
364 FileBrowser fb;
365 fb.unmountSDCard();
366 bool r = fb.mountSDCard(false, "/SD", 1);
367 if (!r) {
368 sdResultLabel->labelStyle().textColor = Color::BrightRed;
369 sdResultLabel->setText("Mount Failed!");
370 if (mcpAvailable)
371 initMCP();
372 return;
373 }
374 // write test
375 fb.setDirectory("/SD");
376 constexpr int size = 16384;
377 bool ok = true;
378 auto fname = fb.createTempFilename();
379 auto f = fopen(fname, "wb");
380 if (f) {
381 for (int i = 0; i < size; ++i)
382 if (fputc(i & 0xff, f) != (i & 0xff)) {
383 ok = false;
384 break;
385 }
386 fclose(f);
387 }
388 if (!f || !ok) {
389 free(fname);
390 sdResultLabel->labelStyle().textColor = Color::BrightRed;
391 sdResultLabel->setText("Write Failed!");
392 return;
393 }
394 // read test
395 f = fopen(fname, "rb");
396 if (f) {
397 for (int i = 0; i < size; ++i)
398 if (fgetc(f) != (i & 0xff)) {
399 ok = false;
400 break;
401 }
402 fclose(f);
403 }
404 free(fname);
405 fb.unmountSDCard();
406 if (mcpAvailable)
407 initMCP();
408 if (!f || !ok) {
409 sdResultLabel->labelStyle().textColor = Color::BrightRed;
410 sdResultLabel->setText("Read Failed!");
411 return;
412 }
413
414 sdResultLabel->labelStyle().textColor = Color::Green;
415 sdResultLabel->setText("Success!");
416 }

427 }
428 } else {
429 wifiResultLabel->labelStyle().textColor = Color::BrightRed;
430 wifiResultLabel->setText("Wifi Scan Failed!");
431 }
432 }
