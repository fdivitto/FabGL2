Sprites

The simplest sprite example: three sprites move along sine waves over a grid background at 320x200.

Each sprite uses a different bitmap pixel format:

Sprite Format
1 PixelFormat::Mask (1 bit per pixel, single color)
2 PixelFormat::RGBA2222 (64 colors + 2 alpha bits)
3 PixelFormat::RGBA8888 (true color + alpha, converted to the display format)

The demo runs on its own and speeds up over time: there are no controls.

What it demonstrates

  • Creating fabgl::Bitmap objects from arrays in the supported pixel formats
  • fabgl::Sprite, VGAController::setSprites() and refreshSprites()
  • Sprites move without damaging the background (the controller saves and restores it)

Hardware

  • VGA output (default pins)

Build

See Building the examples.

91void setup()
92{
93
94 DisplayController.begin();
95 DisplayController.setResolution(VGA_320x200_75Hz);
96
97 // draw background
98 Canvas cv(&DisplayController);
99 cv.setBrushColor(RGB888(0, 0, 64));
100 cv.clear();
101 cv.setPenColor(RGB888(64, 64, 0));
102 for (int i = 0; i < cv.getWidth(); i += 10)
103 cv.drawLine(i, 0, i, cv.getHeight() - 1);
104 for (int i = 0; i < cv.getHeight(); i += 10)
105 cv.drawLine(0, i, cv.getWidth() - 1, i);
106
107 // adjust this to center screen in your monitor
108 //DisplayController.moveScreen(-6, 0);
109
110 // associates bitmaps to sprites
111 sprites[0].addBitmap(&bitmap1);
112 sprites[1].addBitmap(&bitmap2);
113 sprites[2].addBitmap(&bitmap3);
114
115 // sets initial position
116 sprites[0].moveTo(0, 50);
117 sprites[1].moveTo(0, 100);
118 sprites[2].moveTo(0, 150);
119
120 // makes sprites visible
121 sprites[0].visible = true;
122 sprites[1].visible = true;
123 sprites[2].visible = true;
124
125 // add sprites to display controller
126 DisplayController.setSprites(sprites, 3);
127}

131void loop()
132{
133 static double count = 0;
134
135 // set position of sprite 0
136 sprites[0].x += 1;
137 if (sprites[0].x >= DisplayController.getViewPortWidth())
138 sprites[0].x = 0;
139 sprites[0].y = 50 + 35.0 * sin(count);
140
141 // set position of sprite 1
142 sprites[1].x += 1;
143 if (sprites[1].x >= DisplayController.getViewPortWidth())
144 sprites[1].x = 0;
145 sprites[1].y = 100 + 35.0 * cos(count);
146
147 // set position of sprite 2
148 sprites[2].x += 1;
149 if (sprites[2].x >= DisplayController.getViewPortWidth())
150 sprites[2].x = 0;
151 sprites[2].y = 150 + 35.0 * sin(count);
152
153 // update sprites positions
154 DisplayController.refreshSprites();
155
156 count += 0.10;
157
158 vTaskDelay(pdMS_TO_TICKS(std::max(10, 50 - (int)count / 2)));
159}
