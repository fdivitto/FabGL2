81void I2C::end()
82{
83 for (int i = 0; i < m_devicesCount; ++i)
84 i2c_master_bus_rm_device(m_devices[i].handle);
85 m_devicesCount = 0;
86
87 if (m_bus)
88 i2c_del_master_bus(m_bus);
89 m_bus = nullptr;
90
91 if (m_mutex)
92 vSemaphoreDelete(m_mutex);
93 m_mutex = nullptr;
94}

97// m_mutex must be taken
98i2c_master_dev_handle_t I2C::getDevice(int address, int frequency)
99{
100 for (int i = 0; i < m_devicesCount; ++i)
101 if (m_devices[i].address == address && m_devices[i].frequency == frequency)
102 return m_devices[i].handle;
103
104 // remove oldest device if there isn't space
105 if (m_devicesCount == MAXDEVICES) {
106 i2c_master_bus_rm_device(m_devices[0].handle);
107 memmove(m_devices, m_devices + 1, sizeof(DeviceEntry) * (MAXDEVICES - 1));
108 --m_devicesCount;
109 }
110
111 i2c_device_config_t devConfig = { };
112 devConfig.dev_addr_length = I2C_ADDR_BIT_LEN_7;
113 devConfig.device_address = address;
114 devConfig.scl_speed_hz = frequency;
115
116 i2c_master_dev_handle_t handle;
117 if (i2c_master_bus_add_device(m_bus, &devConfig, &handle) != ESP_OK)
118 return nullptr;
119
120 m_devices[m_devicesCount++] = { handle, (uint16_t) address, (uint32_t) frequency };
121 return handle;
122}

125bool I2C::write(int address, uint8_t * buffer, int size, int frequency, int timeOutMS)
126{
127 if (!m_bus)
128 return false;
129 bool ret = false;
130 xSemaphoreTake(m_mutex, portMAX_DELAY);
131 auto dev = getDevice(address, frequency);
132 if (dev)
133 ret = (i2c_master_transmit(dev, buffer, size, timeOutMS) == ESP_OK);
134 xSemaphoreGive(m_mutex);
135 return ret;
136}

139int I2C::read(int address, uint8_t * buffer, int size, int frequency, int timeOutMS)
140{
141 if (!m_bus)
142 return 0;
143 int ret = 0;
144 xSemaphoreTake(m_mutex, portMAX_DELAY);
145 auto dev = getDevice(address, frequency);
146 if (dev && i2c_master_receive(dev, buffer, size, timeOutMS) == ESP_OK)
147 ret = size;
148 xSemaphoreGive(m_mutex);
149 return ret;
150}
