64// 1 = sunday
65void DateTime::calcDayOfWeek()
66{
67 static const int8_t t[] = { 0, 3, 2, 5, 0, 3, 5, 1, 4, 6, 2, 4 };
68 int y = year - (month < 3);
69 dayOfWeek = 1 + (y + y / 4 - y / 100 + y / 400 + t[month - 1] + dayOfMonth) % 7;
70}

127DateTime const & DS3231::datetime()
128{
129 // read 7 registers starting from reg 0
130 uint8_t b = 0;
131 m_i2c->write(DS3231_ADDR, &b, 1, DS3231_FREQ);
132 uint8_t buf[7];
133 m_available = m_i2c->read(DS3231_ADDR, buf, 7, DS3231_FREQ);
134
135 // decode
136 if (m_available) {
137 m_datetime.seconds = (buf[0] & 0x0f) + ((buf[0] & 0x70) >> 4) * 10;
138 m_datetime.minutes = (buf[1] & 0x0f) + ((buf[1] & 0x70) >> 4) * 10;
139 m_datetime.hours = (buf[2] & 0x0f) + ((buf[2] & 0x10) >> 4) * 10;
140 if (buf[2] & (1 << 6)) {
141 // 12 hours mode (convert to 24)
142 if (buf[2] & (1 << 5))
143 m_datetime.hours += 12;
144 } else {
145 // 24 hours mode
146 m_datetime.hours += ((buf[2] & 0x20) >> 4) * 10;
147 }
148 m_datetime.dayOfWeek = buf[3] & 7;
149 m_datetime.dayOfMonth = (buf[4] & 0x0f) + ((buf[4] & 0x30) >> 4) * 10;
150 m_datetime.month = (buf[5] & 0x0f) + ((buf[5] & 0x10) >> 4) * 10;
151 m_datetime.year = (buf[6] & 0x0f) + ((buf[6] & 0xf0) >> 4) * 10 + 2000;
152 }
153 return m_datetime;
154}

157bool DS3231::setDateTime(DateTime const & value)
158{
159 // write 7 registers starting from reg 0
160 uint8_t buf[8];
161 buf[0] = 0; // starting reg address
162 buf[1] = (value.seconds % 10) | ((value.seconds / 10) << 4);
163 buf[2] = (value.minutes % 10) | ((value.minutes / 10) << 4);
164 buf[3] = (value.hours % 10) | ((value.hours / 10) << 4); // bit6=0 -> 24 hours mode
165 buf[4] = value.dayOfWeek;
166 buf[5] = (value.dayOfMonth % 10) | ((value.dayOfMonth / 10) << 4);;
167 buf[6] = (value.month % 10) | ((value.month / 10) << 4);
168 buf[7] = ((value.year - 2000) % 10) | (((value.year - 2000) / 10) << 4);
169 m_i2c->write(DS3231_ADDR, buf, 8, DS3231_FREQ);
170
171 // update oscillator stop flag (make date no more invalid)
172 uint8_t st = readReg(0x0f);
173 writeReg(0x0f, st & 0x7f);
174
175 return m_available;
176}

179double DS3231::temperature()
180{
181 if ((readReg(0x0f) & (0b100)) == 0) {
182 // not busy, force "convert temperature"
183 uint8_t r = readReg(0x0e) | 0x20;
184 writeReg(0x0e, r);
185 // wait for not busy
186 vTaskDelay(2 / portTICK_PERIOD_MS);
187 while ((readReg(0x0f) & (0b100)) != 0 && m_available) {
188 vTaskDelay(1);
189 }
190 }
191
192 // read 2 registers starting from reg 0x11
193 uint8_t b = 0x11;
194 m_i2c->write(DS3231_ADDR, &b, 1, DS3231_FREQ);
195 uint8_t buf[2];
196 m_available = m_i2c->read(DS3231_ADDR, buf, 2, DS3231_FREQ);
197
198 return (int8_t)buf[0] + 0.25 * (buf[1] >> 6);
199}
